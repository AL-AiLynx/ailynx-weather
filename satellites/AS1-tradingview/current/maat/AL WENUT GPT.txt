f_norm100(float v, float lo, float hi) =>
    na(v) or na(lo) or na(hi) ? na : hi > lo ? math.max(0.0, math.min(100.0, (v - lo) / (hi - lo) * 100.0)) : 50.0

f_norm100Log(float v, float lo, float hi) =>
    float out = na
    if not na(v) and not na(lo) and not na(hi)
        float vLog = math.log(math.max(v, 1.0))
        float loLog = math.log(math.max(lo, 1.0))
        float hiLog = math.log(math.max(hi, 1.0))
        if hiLog > loLog
            out := math.max(0.0, math.min(100.0, (vLog - loLog) / (hiLog - loLog) * 100.0))
        else
            out := 50.0
    out

f_normScore(float v, float lo, float hi) =>
    scoreScaleMode == "LOG" ? f_norm100Log(v, lo, hi) : f_norm100(v, lo, hi)

f_arrMin(array<float> arr) =>
    float out = na
    for i = 0 to array.size(arr) - 1
        float v = array.get(arr, i)
        if not na(v)
            if na(out) or v < out
                out := v
    out

f_arrMax(array<float> arr) =>
    float out = na
    for i = 0 to array.size(arr) - 1
        float v = array.get(arr, i)
        if not na(v)
            if na(out) or v > out
                out := v
    out

f_topIndex(array<float> arr) =>
    int idx = 0
    float best = na
    for i = 0 to array.size(arr) - 1
        float v = array.get(arr, i)
        if not na(v)
            if na(best) or v > best
                best := v
                idx := i
    idx
f_topIndexEx(array<float> arr, int ex1, int ex2) =>
    int idx = 0
    float best = na
    for i = 0 to array.size(arr) - 1
        if i != ex1 and i != ex2
            float v = array.get(arr, i)
            if not na(v)
                if na(best) or v > best
                    best := v
                    idx := i
    idx

f_wenJsonPair(string k, string v) =>
    "\"" + k + "\":\"" + v + "\""

f_wenJsonNum(string k, float v) =>
    "\"" + k + "\":\"" + str.tostring(v) + "\""

f_wenJsonInt(string k, int v) =>
    "\"" + k + "\":\"" + str.tostring(v) + "\""

f_wenSymbol() =>
    syminfo.prefix == "" ? syminfo.ticker : syminfo.prefix + ":" + syminfo.ticker

f_wenutJson(string ev, string trg, string defenseType) =>
    string s = "{"
    s += f_wenJsonPair("source", "TradingView")
    s += "," + f_wenJsonPair("system", "AiLynx")
    s += "," + f_wenJsonPair("engine", "WENUT")

    s += "," + f_wenJsonPair("defense_type", defenseType)
    s += "," + f_wenJsonPair("trigger", trg)
    s += "," + f_wenJsonPair("symbol", f_wenSymbol())
    s += "," + f_wenJsonPair("tf", timeframe.period)
