f_isNear(float currentPrice, float level, float pctTol) =>
    bool out = false
    if not na(currentPrice) and not na(level) and currentPrice != 0
        out := math.abs((level - currentPrice) / currentPrice) * 100.0 <= pctTol

f_isSameLevel(float a, float b, float pctTol) =>
    bool out = false
    if not na(a) and not na(b)
        float base = math.max(math.abs(a), 1e-6)
        out := math.abs(a - b) / base * 100.0 <= pctTol
    out

f_bestVolumePriceConfirmed(float zoneLow, float zoneHigh, int bars, int bins, float refPrice) =>
    float bestPrice = na
    float bestVol = -1.0
    if not na(zoneLow) and not na(zoneHigh) and zoneHigh > zoneLow and bars > 0 and bins > 0
        float step = (zoneHigh - zoneLow) / bins
        if step > 0
            float[] vols = array.new_float(bins + 1, 0.0)
            for i = 0 to bars - 1
                float hi_ = high[i + 1]
                float lo_ = low[i + 1]
                float vol_ = volume[i + 1]
                if not na(hi_) and not na(lo_) and not na(vol_)
                    int idxLo = int(math.floor((lo_ - zoneLow) / step))
                    int idxHi = int(math.ceil((hi_ - zoneLow) / step))
                    idxLo := math.max(0, math.min(bins, idxLo))
                    idxHi := math.max(0, math.min(bins, idxHi))
                    int span = math.max(1, idxHi - idxLo + 1)
                    float volShare = vol_ / span
                    for k = idxLo to idxHi
                        array.set(vols, k, array.get(vols, k) + volShare)
            for k = 0 to bins
                float volAt = array.get(vols, k)
                float lvl = zoneLow + step * k
                bool betterVol = volAt > bestVol
                bool betterTie = volAt == bestVol and (na(bestPrice) or math.abs(lvl - refPrice) < math.abs(bestPrice - refPrice))
                if betterVol or betterTie
                    bestVol := volAt
                    bestPrice := lvl
    bestPrice

f_bestTouchPriceConfirmed(float zoneLow, float zoneHigh, int bars, int bins, float refPrice) =>
    float bestPrice = na
    int bestCount = -1
    if not na(zoneLow) and not na(zoneHigh) and zoneHigh > zoneLow and bars > 0 and bins > 0
        float step = (zoneHigh - zoneLow) / bins
        if step > 0
            int[] counts = array.new_int(bins + 1, 0)
            for i = 0 to bars - 1
                float hi_ = high[i + 1]
                float lo_ = low[i + 1]
                if not na(hi_) and not na(lo_)
                    int idxLo = int(math.floor((lo_ - zoneLow) / step))
                    int idxHi = int(math.ceil((hi_ - zoneLow) / step))
                    idxLo := math.max(0, math.min(bins, idxLo))
                    idxHi := math.max(0, math.min(bins, idxHi))
                    for k = idxLo to idxHi
                        array.set(counts, k, array.get(counts, k) + 1)
            for k = 0 to bins
                int cnt = array.get(counts, k)
                float lvl = zoneLow + step * k
                bool betterCount = cnt > bestCount
                bool betterTie = cnt == bestCount and (na(bestPrice) or math.abs(lvl - refPrice) < math.abs(bestPrice - refPrice))
                if betterCount or betterTie
                    bestCount := cnt
                    bestPrice := lvl
    bestPrice

f_touchCountConfirmed(float level, int bars) =>
    float cnt = 0.0
    if not na(level)
        for i = 0 to bars - 1
            float hi_ = high[i + 1]
            float lo_ = low[i + 1]
            if not na(hi_) and not na(lo_) and hi_ >= level and lo_ <= level
                cnt += 1.0
    cnt

f_buildHeru(int len, int bins) =>
    float refClose = close[1]
    float smaV = ta.sma(close, len)[1]
    float emaV = ta.ema(close, len)[1]
    float yHigh = ta.highest(high[1], len)
    float yLow = ta.lowest(low[1], len)
    float pocV = f_bestVolumePriceConfirmed(yLow, yHigh, len, bins, refClose)
    float memV = f_bestTouchPriceConfirmed(yLow, yHigh, len, bins, refClose)
    float memTouches = f_touchCountConfirmed(memV, len)
    [smaV, emaV, pocV, memV, memTouches, refClose]

f_weightedCenter(float p1, float w1, float p2, float w2, float p3, float w3, float p4, float w4) =>
    float sumP = 0.0
    float sumW = 0.0
    if not na(p1)
        sumP += p1 * w1
        sumW += w1
    if not na(p2)
        sumP += p2 * w2
        sumW += w2
    if not na(p3)
        sumP += p3 * w3
        sumW += w3
    if not na(p4)
        sumP += p4 * w4
        sumW += w4
    float out = na
    if sumW > 0
        out := sumP / sumW
    out

f_cluster(float currentPrice, float p1, float w1, string n1, float p2, float w2, string n2, float p3, float w3, string n3, float p4, float w4, string n4, bool wantUpper, float pct) =>
    float[] prices = array.new_float(0)
    float[] weights = array.new_float(0)
    string[] names = array.new_string(0)
    array.push(prices, p1)
    array.push(prices, p2)
    array.push(prices, p3)
    array.push(prices, p4)
    array.push(weights, w1)
    array.push(weights, w2)
    array.push(weights, w3)
    array.push(weights, w4)
    array.push(names, n1)
    array.push(names, n2)
    array.push(names, n3)
    array.push(names, n4)
    float bestPx = na
    float bestWeight = 0.0
    float bestMembers = 0.0
    string bestCombo = "NONE"
    for i = 0 to 3
        float pi = array.get(prices, i)
        bool seedOk = not na(pi) and (wantUpper ? pi > currentPrice : pi < currentPrice)
        if seedOk
            float sumP = 0.0
            float sumW = 0.0
            float members = 0.0
            string combo = ""
            for j = 0 to 3
                float pj = array.get(prices, j)
                float wj = array.get(weights, j)
                string nj = array.get(names, j)
                bool memberSide = not na(pj) and (wantUpper ? pj > currentPrice : pj < currentPrice)
                float dPct = na
                if not na(pj) and pi != 0
                    dPct := math.abs((pj - pi) / pi) * 100.0
                if memberSide and not na(dPct) and dPct <= pct
                    sumP += pj * wj
                    sumW += wj
                    members += 1.0
                    combo := combo == "" ? nj : combo + "+" + nj
            if sumW > 0
                float px = sumP / sumW
                bool betterWeight = sumW > bestWeight
                bool betterMembers = sumW == bestWeight and members > bestMembers
                bool betterDistance = sumW == bestWeight and members == bestMembers and (na(bestPx) or math.abs(px - currentPrice) < math.abs(bestPx - currentPrice))
                if betterWeight or betterMembers or betterDistance
                    bestWeight := sumW
                    bestPx := px
                    bestMembers := members
                    bestCombo := combo
    [bestPx, bestWeight, bestMembers, bestCombo]

f_h365JsonPair(string k, string v) =>
    "\"" + k + "\":\"" + v + "\""

f_h365JsonNum(string k, float v) =>
    "\"" + k + "\":\"" + str.tostring(v) + "\""

f_h365JsonInt(string k, int v) =>
    "\"" + k + "\":\"" + str.tostring(v) + "\""

f_h365Symbol() =>
    syminfo.prefix == "" ? syminfo.ticker : syminfo.prefix + ":" + syminfo.ticker
