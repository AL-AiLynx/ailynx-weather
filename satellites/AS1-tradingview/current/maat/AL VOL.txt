//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// AL VOLUME V2 — 강한 매수세 / 매도세 구분 강화 버전
//
// [V2 개선 사항]
// FIX-1 : Delta 기반 봉 내부 매수/매도 압력 분해 (Bid/Ask Split Estimation)
// FIX-2 : Body Ratio — 봉 형태 기반 방향 강도 측정
// FIX-3 : Wick Pressure — 위/아래 꼬리 압력 비율
// FIX-4 : CVD (Cumulative Volume Delta) — 누적 순매수 압력
// FIX-5 : Absorption 탐지 — 큰 거래량 + 작은 봉 = 세력 흡수/분산
// FIX-6 : Climax 탐지 — 극단 거래량 후 반전 경고
// FIX-7 : bearSnort 추가 — 강한 매도세 대칭 탐지
// FIX-8 : 강도 분류 5단계 (STRONG_BUY / BUY / NEUTRAL / SELL / STRONG_SELL)
// FIX-9 : 흡수/클라이맥스 hidden output 추가
// FIX-10: plot count 64 제한 회피 — 마커 plotshape 제거, 배경 bgcolor 1개로 통합
// FIX-11: 상단 상태줄 입력값 숨김 + 오른쪽 거래량 스케일 M 단위 압축
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// Groups
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
grpCore    = "01. 기본 설정"
grpDelta   = "02. Delta / CVD 설정"
grpBody    = "03. 봉 형태 압력 설정"
grpAbsorb  = "04. 흡수 / 클라이맥스 설정"
grpLogic   = "05. 파란 거래량 로직"
grpDry     = "06. 드라이 / 파워 거래량"
grpColor   = "07. 색상 설정"
grpLine    = "08. 평균선 / 기준선"
grpMark    = "09. 마커 / 배경"
grpTable   = "10. 정보표"
grpLabel   = "11. 라벨"
grpOutput  = "12. 출력값 / 알림"

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 01. Core
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
vol = nz(volume)

pocketLookback = input.int(10,  "포켓 피봇 기준 봉 수",             minval=1, maxval=200, group=grpCore, display=display.none)
avgLen         = input.int(50,  "거래량 평균 기준 봉 수",            minval=1, maxval=500, group=grpCore, display=display.none)
relLen         = input.int(20,  "상대 거래량 기준 봉 수",            minval=1, maxval=500, group=grpCore, display=display.none)
upDownLen      = input.int(20,  "상승 / 하락 거래량 비율 기준 봉 수", minval=1, maxval=500, group=grpCore, display=display.none)
bullSnortLen   = input.int(50,  "강한 매수 / 매도 거래량 평균 봉 수", minval=1, maxval=500, group=grpCore, display=display.none)
turnoverMode   = input.string("거래량", "파워 기준 계산 방식", options=["거래량", "거래대금"], group=grpCore, display=display.none)
usePrevColor   = input.bool(false, "이전 봉 방향 기준으로 색상 판단", group=grpCore, display=display.none)

isUp   = usePrevColor ? close >= close[1] : close >= open
isDown = not isUp

volAvg  = ta.sma(vol, avgLen)
relAvg  = ta.sma(vol, relLen)
relVol  = relAvg > 0.0 ? vol / relAvg : 0.0

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// [FIX-1] Delta 기반 봉 내부 매수/매도 압력 분해
// Bid/Ask Split Estimation (Tick Rule 대안 — Pine Script 가능 버전)
// 원리: (close - low) / (high - low) 비율이 클수록 봉 내에서 매수 우위
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
useDelta    = input.bool(true,  "Delta 압력 계산 사용",          group=grpDelta, display=display.none)
deltaMode   = input.string("정밀", "Delta 계산 방식", options=["단순", "정밀"], group=grpDelta, display=display.none)
cvdLen      = input.int(50,    "CVD 단기 누적 기준 봉 수",       minval=5, maxval=500, group=grpDelta, display=display.none)
cvdNormLen  = input.int(200,   "CVD 정규화 기준 봉 수",          minval=20, maxval=1000, group=grpDelta, display=display.none)
showCvd     = input.bool(true,  "CVD 라인 표시",                 group=grpDelta, display=display.none)
cvdLineColor = input.color(color.new(color.yellow, 0), "CVD 라인 색상", group=grpDelta, display=display.none)

// [핵심 공식] 봉 내부 매수/매도 비율 추정
rangeBar    = high - low
safeRange   = math.max(rangeBar, syminfo.mintick)

// 단순: 봉 방향으로만 분기
// 정밀: 봉 내에서 close 위치 비율로 매수/매도 추정
buyPressureRaw = deltaMode == "정밀" ? (close - low) / safeRange * vol : (isUp ? vol : 0.0)

sellPressureRaw = deltaMode == "정밀" ? (high - close) / safeRange * vol : (isDown ? vol : 0.0)

// 봉별 Delta (순매수: 양수 = 매수 우위, 음수 = 매도 우위)
deltaBar = buyPressureRaw - sellPressureRaw

// CVD — 단기 누적 순매수 압력
cvdRolling = math.sum(deltaBar, cvdLen)

// CVD 정규화 (z-score 방식 — 극단값 방어)
cvdMean   = ta.sma(cvdRolling, cvdNormLen)
cvdStdev  = ta.stdev(cvdRolling, cvdNormLen)
cvdZ      = cvdStdev > 0 ? (cvdRolling - cvdMean) / cvdStdev : 0.0

// CVD 방향 판단 (Z-score 기준)
cvdBull   = cvdZ >= 0.5     // 누적 매수 우위
cvdBear   = cvdZ <= -0.5    // 누적 매도 우위
cvdStrong = math.abs(cvdZ) >= 1.2  // 강한 방향 확립

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// [FIX-2] Body Ratio — 봉 형태 기반 방향 강도
// 원리: body가 전체 범위 대비 클수록 방향성이 강한 봉
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
useBodyFilter  = input.bool(true,  "봉 형태 필터 사용",          group=grpBody, display=display.none)
strongBodyRatio = input.float(0.60, "강한 방향봉 Body 비율 기준", minval=0.2, maxval=0.95, step=0.05, group=grpBody, display=display.none)
weakBodyRatio   = input.float(0.25, "약한 방향봉 Body 비율 기준", minval=0.05, maxval=0.5, step=0.05, group=grpBody, display=display.none)

bodySize  = math.abs(close - open)
bodyRatio = safeRange > 0 ? bodySize / safeRange : 0.5  // 0~1

// 꼬리 압력
wickUp   = high - math.max(close, open)
wickDown = math.min(close, open) - low
wickUpRatio   = safeRange > 0 ? wickUp   / safeRange : 0.0
wickDownRatio = safeRange > 0 ? wickDown / safeRange : 0.0

// 꼬리 거부 압력 (위꼬리 = 매도 거부, 아래꼬리 = 매수 거부)
// 상승봉에서 위꼬리가 크면 → 매도 저항 존재
// 하락봉에서 아래꼬리가 크면 → 매수 지지 존재
wickRejectionUp   = isUp   and wickUpRatio   >= 0.40  // 상승봉인데 매도 압력
wickRejectionDown = isDown and wickDownRatio >= 0.40  // 하락봉인데 매수 지지

// 강한 방향봉 / 약한 방향봉
strongBody = bodyRatio >= strongBodyRatio
weakBody   = bodyRatio <= weakBodyRatio

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// [FIX-3] 기존 upVol/downVol — Delta 방식으로 업그레이드
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 기존: upVol = isUp ? vol : 0
// 개선: buyPressureRaw / sellPressureRaw 사용
upVol   = buyPressureRaw    // 봉 내 매수 추정 거래량
downVol = sellPressureRaw   // 봉 내 매도 추정 거래량

upVolSum   = math.sum(upVol,   upDownLen)
downVolSum = math.sum(downVol, upDownLen)

// 상승/하락 거래량 비율 (개선됨 — Delta 기반)
upDownRatio = downVolSum > 0.0 ? upVolSum / downVolSum : (upVolSum > 0.0 ? 99.0 : 0.0)

// EMA 가중 비율 (최근성 반영 — V2 신규)
upVolEma   = ta.ema(upVol,   upDownLen)
downVolEma = ta.ema(downVol, upDownLen)
upDownEmaRatio = downVolEma > 0.0 ? upVolEma / downVolEma : (upVolEma > 0.0 ? 99.0 : 0.0)

turnoverValue = turnoverMode == "거래대금" ? vol * close : vol

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// [FIX-4] Absorption 탐지
// 원리: 평균 대비 큰 거래량 + 작은 봉 = 세력이 물량을 조용히 흡수/분산 중
// 매수 흡수: 하락 중 하락봉인데 아래꼬리 크고 거래량 큼 → 세력 매수
// 매도 흡수: 상승 중 상승봉인데 위꼬리 크고 거래량 큼 → 세력 매도
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
showAbsorb     = input.bool(true,  "흡수봉 탐지 표시",                  group=grpAbsorb, display=display.none)
absorbVolMult  = input.float(1.5,  "흡수봉 거래량 기준 (평균의 N배)",   minval=1.0, step=0.1, group=grpAbsorb, display=display.none)
absorbBodyMax  = input.float(0.35, "흡수봉 최대 Body 비율",             minval=0.1, maxval=0.6, step=0.05, group=grpAbsorb, display=display.none)
absorbWickMin  = input.float(0.35, "흡수봉 최소 Wick 비율",             minval=0.1, maxval=0.7, step=0.05, group=grpAbsorb, display=display.none)

// 매수 흡수: 큰 거래량 + 작은 봉 + 아래꼬리 (매수세가 조용히 받아내는 중)
buyAbsorption = vol >= volAvg * absorbVolMult and bodyRatio <= absorbBodyMax and wickDownRatio >= absorbWickMin

// 매도 흡수: 큰 거래량 + 작은 봉 + 위꼬리 (매도세가 조용히 내다 파는 중)
sellAbsorption = vol >= volAvg * absorbVolMult and bodyRatio <= absorbBodyMax and wickUpRatio >= absorbWickMin

// 중립 흡수: 큰 거래량 + 매우 작은 봉 (양방향 균형)
neutralAbsorption = vol >= volAvg * absorbVolMult and bodyRatio <= absorbBodyMax * 0.5 and not buyAbsorption and not sellAbsorption

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// [FIX-5] Climax 탐지
// 원리: 극단 거래량 후 반전 가능성 경고
// 매수 클라이맥스: 강한 상승 지속 후 역대급 거래량 → 천장 가능성
// 매도 클라이맥스: 강한 하락 지속 후 역대급 거래량 → 바닥 가능성
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
showClimax       = input.bool(true, "클라이맥스 탐지 표시",                 group=grpAbsorb, display=display.none)
climaxLookback   = input.int(50,    "클라이맥스 기준 기간 (봉 수)",          minval=10, maxval=200, group=grpAbsorb, display=display.none)
climaxVolPct     = input.float(0.90,"클라이맥스 거래량 기준 (역대 최고의 N)", minval=0.5, maxval=1.0, step=0.05, group=grpAbsorb, display=display.none)
climaxTrendLen   = input.int(5,     "클라이맥스 방향 확인 봉 수",            minval=2, maxval=20, group=grpAbsorb, display=display.none)

climaxHighVol = vol >= ta.highest(vol, climaxLookback) * climaxVolPct

// 방향 지속 확인 (같은 방향 봉이 N개 이상 연속)
upStreak   = 0
downStreak = 0
for i = 1 to climaxTrendLen
    if close[i] >= open[i]
        upStreak   += 1
    if close[i] < open[i]
        downStreak += 1

buyClimax  = climaxHighVol and isUp   and upStreak   >= climaxTrendLen - 1
sellClimax = climaxHighVol and isDown and downStreak >= climaxTrendLen - 1

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// [FIX-6] 강한 매수 / 매도세 5단계 강도 분류
// STRONG_BUY / BUY / NEUTRAL / SELL / STRONG_SELL
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
strongBuyTh   = input.float(1.8, "강한 매수 상대 거래량 기준",  minval=1.0, step=0.1, group=grpDry, display=display.none)
strongSellTh  = input.float(1.8, "강한 매도 상대 거래량 기준",  minval=1.0, step=0.1, group=grpDry, display=display.none)
highRelTh     = input.float(1.4, "일반 강세 상대 거래량 기준",  minval=1.0, step=0.1, group=grpDry, display=display.none)

// [FIX-7] bearSnort (강한 매도 마커) — 기존 bullSnort와 대칭
bullSnortAvg  = ta.sma(upVol,   bullSnortLen)
bearSnortAvg  = ta.sma(downVol, bullSnortLen)
bullSnortTh   = input.float(1.8, "강한 매수 마커 상대 기준", minval=1.0, step=0.1, group=grpDry, display=display.none)
bearSnortTh   = input.float(1.8, "강한 매도 마커 상대 기준", minval=1.0, step=0.1, group=grpDry, display=display.none)

bullSnort = isUp and bullSnortAvg > 0.0 and upVol >= bullSnortAvg * bullSnortTh and (not useBodyFilter or strongBody)  // [FIX-2] Body 필터 추가

bearSnort = isDown and bearSnortAvg > 0.0 and downVol >= bearSnortAvg * bearSnortTh and (not useBodyFilter or strongBody)  // [FIX-2] Body 필터 추가

// [FIX-8] 5단계 강도 판단
// 조건: 상대 거래량 + Delta 방향 + Body 강도 + CVD 방향 통합
strongBuy = isUp and relVol >= strongBuyTh and strongBody and (not useDelta or deltaBar > 0) and (not useDelta or cvdBull or cvdStrong)

strongSell = isDown and relVol >= strongSellTh and strongBody and (not useDelta or deltaBar < 0) and (not useDelta or cvdBear or cvdStrong)
// 중간 강도
midBuy  = isUp   and relVol >= highRelTh and not strongBuy
midSell = isDown and relVol >= highRelTh and not strongSell

// Wick 거부 신호 (방향과 반대 압력)
rejectionUp   = isUp   and wickRejectionUp   and relVol >= highRelTh   // 상승봉인데 매도 저항
rejectionDown = isDown and wickRejectionDown and relVol >= highRelTh   // 하락봉인데 매수 지지

// 강도 코드 (숫자 → hidden output 용)
// 4=STRONG_BUY, 3=BUY, 2=NEUTRAL, 1=SELL, 0=STRONG_SELL
pressureCode = strongBuy ? 4 : midBuy ? 3 : strongSell ? 0 : midSell ? 1 : 2

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 02 (기존). Blue Volume Logic
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
usePocketPivot  = input.bool(true,  "포켓 피봇 거래량 사용",        group=grpLogic, display=display.none)
useHighestVol   = input.bool(true,  "기간 최고 거래량 사용",         group=grpLogic, display=display.none)
useRelBlue      = input.bool(true,  "상대 거래량 파란 막대 사용",    group=grpLogic, display=display.none)
usePowerBlue    = input.bool(false, "파워 거래량도 파란 막대에 포함", group=grpLogic, display=display.none)

highestLookbackE = input.int(500, "HVE 기준 봉 수", minval=10, maxval=5000, group=grpLogic, display=display.none)
highestLookbackY = input.int(365, "HVY 기준 봉 수", minval=10, maxval=5000, group=grpLogic, display=display.none)
highestLookbackQ = input.int(90,  "HVQ 기준 봉 수", minval=10, maxval=5000, group=grpLogic, display=display.none)
relBlueTh        = input.float(2.0,"파란 막대 상대 거래량 기준",    minval=0.1, step=0.1, group=grpLogic, display=display.none)
blueOnlyUp       = input.bool(false,"파란 최고 거래량을 상승봉에서만 표시", group=grpLogic, display=display.none)
blueNeedScore    = input.int(1,    "파란 막대 필요 점수",           minval=1, maxval=6, group=grpLogic, display=display.none)

downVolPrev    = close[1] < open[1] ? vol[1] : 0.0

f_fmtVol(v) =>
    txt = str.tostring(v, "#.##")
    if v >= 1000000000000.0
        txt := str.tostring(v / 1000000000000.0, "#.##") + "T"
    else if v >= 1000000000.0
        txt := str.tostring(v / 1000000000.0, "#.##") + "B"
    else if v >= 1000000.0
        txt := str.tostring(v / 1000000.0, "#.##") + "M"
    else if v >= 1000.0
        txt := str.tostring(v / 1000.0, "#.##") + "K"
    txt
