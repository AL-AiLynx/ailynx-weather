f_clamp(float v, float lo, float hi) =>
    math.max(lo, math.min(hi, v))

f_linked(float v) =>
    not na(v) and math.abs(v - close) > 0.000001

f_linked4(float d, float s, float r, float e) =>
    f_linked(d) or f_linked(s) or f_linked(r) or f_linked(e)

f_valid(float d, float s, float r, float e) =>
    math.abs(nz(d, 99)) <= 1.5 and nz(s, 999) >= 0 and nz(s, 999) <= 100 and nz(r, 999) >= 0 and nz(r, 999) <= 5 and nz(e, 999) >= 0 and nz(e, 999) <= 9

f_dir(float d, bool ok) =>
    ok ? int(f_clamp(math.round(d), -1, 1)) : 0

f_score(float s, bool ok) =>
    ok ? f_clamp(s, 0, 100) : 50.0

f_risk(float r, bool ok) =>
    ok ? int(f_clamp(math.round(r), 0, 5)) : 0

f_event(float e, bool ok) =>
    ok ? int(f_clamp(math.round(e), 0, 9)) : 0

f_num(float v) =>
    str.tostring(int(math.round(v)))

f_rawNum(float v) =>
    string txt = "na"
    if not na(v)
        txt := str.tostring(v)
    txt

f_pos(string opt) =>
    p = position.top_right
    if opt == "좌측 상단"
        p := position.top_left
    else if opt == "중앙 상단"
        p := position.top_center
    else if opt == "우측 상단"
        p := position.top_right
    else if opt == "좌측 중앙"
        p := position.middle_left
    else if opt == "정중앙"
        p := position.middle_center
    else if opt == "우측 중앙"
        p := position.middle_right
    else if opt == "좌측 하단"
        p := position.bottom_left
    else if opt == "중앙 하단"
        p := position.bottom_center
    else if opt == "우측 하단"
        p := position.bottom_right
    p

f_sizeFromScale(int scale) =>
    s = size.small
    if scale <= 2
        s := size.tiny
    else if scale <= 5
        s := size.small
    else if scale <= 7
        s := size.normal
    else if scale <= 9
        s := size.large
    else
        s := size.huge
    s

f_dirCode(int d) =>
    string txt = "N"
    if d > 0
        txt := "B"
    else if d < 0
        txt := "S"
    txt

f_dirText(int d, bool ok, string lang) =>
    string txt = "MISS"
    if not ok
        if lang == "KR"
            txt := "연결확인"
        else if lang == "CODE"
            txt := "M"
        else
            txt := "MISS"
    else
        if lang == "KR"
            if d > 0
                txt := "상승"
            else if d < 0
                txt := "하락"
            else
                txt := "중립"
        else if lang == "CODE"
            txt := f_dirCode(d)
        else
            if d > 0
                txt := "BULL"
            else if d < 0
                txt := "BEAR"
            else
                txt := "NEUT"
    txt

f_sensorRowText(string sensorKey, bool linked, bool ok, int d, float s, int r, int eventCode, float rawD, float rawS, float rawR, float rawE, string mode, string lang, bool compactMode) =>
    bool codeLike = mode == "GPT" or mode == "COMPACT" or lang == "CODE"
    string eventLang = mode == "GPT" ? "CODE" : lang
    string eventTxt = f_eventText(sensorKey, eventCode, eventLang, compactMode or mode == "GPT" or mode == "COMPACT")
    string txt = ""
    if mode == "DEBUG"
        if ok
            txt := "DIR " + str.tostring(d) + " / S " + f_num(s) + " / R " + str.tostring(r) + " / E " + str.tostring(eventCode) + " / " + eventTxt
        else if linked
            txt := "MISS / DIR " + f_rawNum(rawD) + " / S " + f_rawNum(rawS) + " / R " + f_rawNum(rawR) + " / E " + f_rawNum(rawE)
        else
            txt := "OFF"
    else if not ok
        if codeLike
            txt := linked ? "MISS" : "OFF"
        else if lang == "KR"
            txt := linked ? "연결확인" : "OFF"
        else
            txt := linked ? "MISS" : "OFF"
    else if mode == "GPT"
        txt := f_dirCode(d) + f_num(s) + " R" + str.tostring(r) + " E" + str.tostring(eventCode)
    else if mode == "COMPACT"
        txt := f_dirCode(d) + f_num(s) + " R" + str.tostring(r) + " " + eventTxt
    else
        if compactMode
            txt := f_dirText(d, ok, lang) + " " + f_num(s) + " R" + str.tostring(r) + " E" + str.tostring(eventCode) + " " + eventTxt
        else if lang == "KR"
            txt := f_dirText(d, ok, lang) + " " + f_num(s) + " / 위험 " + str.tostring(r) + " / 이벤트 " + str.tostring(eventCode) + " " + eventTxt
        else if lang == "CODE"
            txt := f_dirText(d, ok, lang) + f_num(s) + " R" + str.tostring(r) + " E" + str.tostring(eventCode) + " " + eventTxt
        else
            txt := f_dirText(d, ok, lang) + " " + f_num(s) + " / RISK " + str.tostring(r) + " / EVENT " + str.tostring(eventCode) + " " + eventTxt
    txt
