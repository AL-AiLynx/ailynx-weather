f_htfDir() =>
    float hEmaFast = ta.ema(close, emaMajorFastLen)
    float hEmaSlow = ta.ema(close, emaMajorSlowLen)
    float hRsi = ta.rsi(close, 14)
    [hPdi, hMdi, hAdx] = ta.dmi(dmiLen, dmiSmooth)
    int result = 0
    if hEmaFast > hEmaSlow and hRsi > 52.0 and hPdi > hMdi and hAdx >= 15.0
        result := 1

float atrV = ta.atr(atrLen)
float atrPct = atrV / math.max(close, syminfo.mintick) * 100.0
float emaFast = ta.ema(close, emaFastLen)
float emaSlow = ta.ema(close, emaSlowLen)
float ema50 = ta.ema(close, emaMajorFastLen)
float ema200 = ta.ema(close, emaMajorSlowLen)
float rsi14 = ta.rsi(close, 14)
float volSma = ta.sma(nz(volume, 0.0), volLen)
float volRatio = volSma > 0.0 ? nz(volume, 0.0) / volSma : 1.0
[pdiV, mdiV, adxV] = ta.dmi(dmiLen, dmiSmooth)

bool atrFilterOK = atrPct >= minAtrPct

// [3-1] Shield: EMA50/200 + RSI + ADX + ATR filter.
float shieldLongBits = 0.0
shieldLongBits += ema50 > ema200 ? 0.30 : 0.0
shieldLongBits += close > ema50 ? 0.20 : 0.0
shieldLongBits += rsi14 > 52.0 ? 0.20 : 0.0
shieldLongBits += pdiV > mdiV ? 0.15 : 0.0
shieldLongBits += adxV >= 20.0 ? 0.15 : 0.0

float shieldShortBits = 0.0
shieldShortBits += ema50 < ema200 ? 0.30 : 0.0
shieldShortBits += close < ema50 ? 0.20 : 0.0
shieldShortBits += rsi14 < 48.0 ? 0.20 : 0.0
shieldShortBits += mdiV > pdiV ? 0.15 : 0.0
shieldShortBits += adxV >= 20.0 ? 0.15 : 0.0
float shieldStructureBias = ema50 > ema200 ? 0.18 : ema50 < ema200 ? -0.18 : 0.0
float shieldNorm = f_clamp((atrFilterOK ? shieldLongBits - shieldShortBits : 0.0) + shieldStructureBias, -1.0, 1.0)

// [3-2] Armor: structure break with volume and pivot confirmation.
float structHi = ta.highest(high[1], armorStructLen)
float structLo = ta.lowest(low[1], armorStructLen)
float pivH = ta.pivothigh(high, pivotLen, pivotLen)
float pivL = ta.pivotlow(low, pivotLen, pivotLen)
var float lastPivH = na
var float lastPivL = na
if not na(pivH)
    lastPivH := pivH
if not na(pivL)
    lastPivL := pivL

bool armorUpBreakRaw = not na(structHi) and close > structHi + atrV * armorBreakAtr
bool armorDnBreakRaw = not na(structLo) and close < structLo - atrV * armorBreakAtr
bool armorUpConfirmed = armorUpBreakRaw and volRatio >= armorVolNeed and (na(lastPivH) or close > lastPivH)
bool armorDnConfirmed = armorDnBreakRaw and volRatio >= armorVolNeed and (na(lastPivL) or close < lastPivL)

var int armorSide = 0
var float armorLevel = na
var int armorStartBar = na
if armorUpConfirmed
    armorSide := 1
    armorLevel := structHi
    armorStartBar := bar_index
else if armorDnConfirmed
    armorSide := -1
    armorLevel := structLo
    armorStartBar := bar_index
else if armorSide == 1 and close < nz(armorLevel, close) - atrV
    armorSide := 0
else if armorSide == -1 and close > nz(armorLevel, close) + atrV
    armorSide := 0

float armorHold = armorSide != 0 and not na(armorStartBar) ? f_clamp01(float(bar_index - armorStartBar) / 20.0) : 0.0
float armorDist = armorSide == 1 and not na(armorLevel) ? f_clamp01((close - armorLevel) / math.max(atrV * 2.0, syminfo.mintick)) : armorSide == -1 and not na(armorLevel) ? f_clamp01((armorLevel - close) / math.max(atrV * 2.0, syminfo.mintick)) : 0.0
float armorPower = f_clamp01(0.40 * f_clamp01((volRatio - 1.0) / 1.0) + 0.35 * armorDist + 0.25 * armorHold)
float armorNorm = float(armorSide) * armorPower

// [3-3] Helmet: sweep return state and session VWAP.
bool newSession = ta.change(time("D")) != 0
var float sessPV = na
var float sessV = na
float barVol = nz(volume, 0.0)
if barstate.isfirst or newSession
    sessPV := hlc3 * barVol
