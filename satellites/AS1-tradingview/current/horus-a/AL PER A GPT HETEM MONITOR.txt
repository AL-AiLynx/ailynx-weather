f_pos(string opt) =>
    opt == "TL" ? position.top_left :
     opt == "TC" ? position.top_center :
     opt == "TR" ? position.top_right :
     opt == "ML" ? position.middle_left :
     opt == "MC" ? position.middle_center :
     opt == "MR" ? position.middle_right :
     opt == "BL" ? position.bottom_left :
     opt == "BC" ? position.bottom_center : position.bottom_right

f_tsize(string opt) =>
    opt == "XS" ? size.tiny : opt == "S" ? size.small : opt == "M" ? size.normal : opt == "L" ? size.large : size.huge

f_tablePos(string opt) =>
    opt == "좌상" ? position.top_left :
     opt == "상중" ? position.top_center :
     opt == "우상" ? position.top_right :
     opt == "좌중" ? position.middle_left :
     opt == "정중" ? position.middle_center :
     opt == "우중" ? position.middle_right :
     opt == "좌하" ? position.bottom_left :
     opt == "하중" ? position.bottom_center : position.bottom_right

f_textSizeFromScale(int sc) =>
    sz = size.normal
    if sc <= 2
        sz := size.tiny
    else if sc <= 4
        sz := size.small
    else if sc <= 6
        sz := size.normal
    else if sc <= 8
        sz := size.large
    else
        sz := size.huge
    sz

f_numTxt(float v) =>
    str.tostring(int(math.round(v)))

f_boolTxt(bool b) =>
    b ? "YES" : "NO"

f_maxMaybe(float a, float b) =>
    na(a) ? b : na(b) ? a : math.max(a, b)

f_minMaybe(float a, float b) =>
    na(a) ? b : na(b) ? a : math.min(a, b)

f_clamp(float x, float lo, float hi) =>
    math.max(lo, math.min(hi, x))

f_dirTxt(int d) =>
