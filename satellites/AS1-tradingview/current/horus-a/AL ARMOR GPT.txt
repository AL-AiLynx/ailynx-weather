string GROUP_CORE = "갑옷 G | 핵심"
int   volLen      = input.int(20, "거래량 SMA 길이", minval=1, group=GROUP_CORE, display=display.none)
float spikeX      = input.float(2.0, "스파이크 배수", minval=1.0, step=0.1, group=GROUP_CORE, display=display.none)
int   smoothLen   = input.int(12, "부드럽게 EMA 길이", minval=1, group=GROUP_CORE, display=display.none)
float decayAlpha  = input.float(0.06, "드리프트 (0~0.2)", minval=0.0, maxval=0.2, step=0.01, group=GROUP_CORE, display=display.none)
float kInsideDamp = input.float(0.65, "K 감쇠 (밴드 내부)", minval=0.2, maxval=1.0, step=0.05, group=GROUP_CORE, display=display.none)

string GROUP_SPIKE = "갑옷 G | 스파이크 필터"
int spikeHiLB      = input.int(30, "강한 스파이크 최고 거래량 범위", minval=5, maxval=300, group=GROUP_SPIKE, display=display.none)
int spikeCD        = input.int(10, "강한 스파이크 쿨다운 (봉)", minval=0, maxval=200, group=GROUP_SPIKE, display=display.none)

string GROUP_BAND = "갑옷 G | 밴드 유효성"
float minBandAtrMult = input.float(0.35, "최소 밴드 폭 (ATR x)", minval=0.05, maxval=3.0, step=0.05, group=GROUP_BAND, display=display.none)

string GROUP_STAB = "갑옷 G | 안정성"
float posEnter    = input.float(0.40, "밴드 내부 진입 기준", minval=0.10, maxval=0.90, step=0.05, group=GROUP_STAB, display=display.none)
float posExit     = input.float(0.25, "밴드 내부 이탈 기준", minval=0.05, maxval=0.80, step=0.05, group=GROUP_STAB, display=display.none)
int   holdBars    = input.int(2, "최소 유지 봉 수", minval=0, maxval=20, group=GROUP_STAB, display=display.none)

string GROUP_RES = "갑옷 G | 저항값"
int kStrong      = input.int(80, "강한 K", minval=50, maxval=100, group=GROUP_RES, display=display.none)
int kMid         = input.int(55, "중간 K", minval=20, maxval=99, group=GROUP_RES, display=display.none)

//==================================================
// [2] 헬퍼
//==================================================
f_clamp01(_x) =>
    math.max(0.0, math.min(1.0, _x))

f_clampInt(_x, _lo, _hi) =>
    math.max(_lo, math.min(_hi, _x))

f_tablePos(_opt) =>
    pos = position.top_right
    if _opt == "TL"
        pos := position.top_left
    else if _opt == "TC"
        pos := position.top_center
    else if _opt == "TR"
        pos := position.top_right
    else if _opt == "ML"
        pos := position.middle_left
    else if _opt == "MC"
        pos := position.middle_center
    else if _opt == "MR"
        pos := position.middle_right
    else if _opt == "BL"
        pos := position.bottom_left
    else if _opt == "BC"
        pos := position.bottom_center
    else
        pos := position.bottom_right
    pos

f_textSize(_opt) =>
    sz = size.small
    if _opt == "TINY"
        sz := size.tiny
    else if _opt == "SMALL"
        sz := size.small
    else if _opt == "NORMAL"
        sz := size.normal
    else
        sz := size.large
    sz

f_dirCode(_dir) =>
    string out = "FLAT"
    if _dir > 0
        out := "UP"
    else if _dir < 0
        out := "DOWN"
    else
        out := "FLAT"
    out

f_stateCode(_state) =>
    string out = "INSIDE"
    if _state == 2
        out := "UP_BRK"
    else if _state == 1
        out := "UP_HOLD"
    else if _state == -1
        out := "DN_HOLD"
    else if _state == -2
        out := "DN_BRK"
    else
        out := "INSIDE"
    out

f_evtCode(_evt) =>
    string out = "NONE"
    if _evt == 10
        out := "SPIKE"
    else if _evt == 20
        out := "UP_BREAK"
    else if _evt == 21
        out := "DN_BREAK"
    else if _evt == 30
        out := "INSIDE"
    else
        out := "NONE"
    out

f_note(_bandOK, _state, _res, _evt) =>
    string out = "WAIT"
    if not _bandOK
        out := "BAND OFF"
    else if _evt == 10
        out := "ANCHOR RESET"
    else if _state == 2 and _res >= 4
        out := "UP STRONG"
    else if _state == -2 and _res >= 4
        out := "DN STRONG"
    else if _state == 1 or _state == -1
        out := "HOLD ZONE"
    else
        out := "INSIDE BAND"
    out

//==================================================
// [3] 핵심 계산
//==================================================
float v    = nz(volume, 0.0)
float vSma = ta.sma(v, volLen)
float vR   = (not na(vSma) and vSma > 0.0) ? (v / vSma) : 1.0
float atrV = ta.atr(14)

float hiVolPrev = ta.highest(volume[1], spikeHiLB)
bool byMult = vR >= spikeX
bool byHigh = (not na(hiVolPrev)) and (volume > hiVolPrev)

var int lastSpikeBar = na
bool cdOK = spikeCD <= 0 ? true : (na(lastSpikeBar) or (bar_index - lastSpikeBar >= spikeCD))

bool isSpikeHard = byMult and byHigh and cdOK
if isSpikeHard
    lastSpikeBar := bar_index

var float anchorRes = na
var float anchorSup = na
if isSpikeHard
    anchorRes := high
    anchorSup := low

float driftRes = na(anchorRes) ? na : (anchorRes + (close - anchorRes) * decayAlpha)
float driftSup = na(anchorSup) ? na : (anchorSup + (close - anchorSup) * decayAlpha)

float resLine = ta.ema(nz(driftRes, close), smoothLen)
float supLine = ta.ema(nz(driftSup, close), smoothLen)

bool hasBand = not na(anchorRes) and not na(anchorSup) and not na(resLine) and not na(supLine)
bool bandOK0 = hasBand and (resLine >= supLine)
float bandW = bandOK0 ? (resLine - supLine) : 0.0
bool bandWideOK = bandOK0 and (bandW >= atrV * minBandAtrMult)
bool bandOK = bandWideOK

int armorDir = 0
int stateRaw = 0
if bandOK
    if close > resLine
        armorDir := 1
        stateRaw := 2
    else if close < supLine
        armorDir := -1
        stateRaw := -2
    else
        armorDir := 0
        float mid = (resLine + supLine) * 0.5
        float halfW = math.max(1e-6, (resLine - supLine) * 0.5)
        float pos = (close - mid) / halfW
        pos := math.max(-1.0, math.min(1.0, pos))
        if pos > posEnter
            stateRaw := 1
        else if pos < -posEnter
            stateRaw := -1
        else
            stateRaw := 0
else
    armorDir := 0
    stateRaw := 0

var int armorState = 0
var int holdLeft = 0
bool isInside = bandOK and (stateRaw == 1 or stateRaw == 0 or stateRaw == -1)
bool isBreak = stateRaw == 2 or stateRaw == -2

if isBreak
    armorState := stateRaw
    holdLeft := 0
else
    if isInside
        float mid2 = (resLine + supLine) * 0.5
        float halfW2 = math.max(1e-6, (resLine - supLine) * 0.5)
        float pos2 = (close - mid2) / halfW2
        pos2 := math.max(-1.0, math.min(1.0, pos2))

        int desired = 0
        if armorState == 1
            desired := pos2 > posExit ? 1 : 0
        else if armorState == -1
            desired := pos2 < -posExit ? -1 : 0
        else
            desired := stateRaw

        if desired != armorState
            holdLeft := holdBars

        if holdLeft > 0
            holdLeft -= 1
            if holdLeft == 0
                armorState := desired
        else
            armorState := desired
    else
        armorState := 0
        holdLeft := 0

float volPart = f_clamp01((vR - spikeX) / math.max(1e-6, spikeX))
float upGap = 0.0
float dnGap = 0.0
if bandOK and atrV > 0
    upGap := math.max(0.0, (close - resLine) / atrV)
    dnGap := math.max(0.0, (supLine - close) / atrV)

float outPart = f_clamp01(math.max(upGap, dnGap) / 2.0)
bool insideBand = bandOK and (close <= resLine) and (close >= supLine)
float damp = insideBand ? kInsideDamp : 1.0

float kF = 100.0 * (0.65 * volPart + 0.35 * outPart) * damp
int armorK = int(math.round(kF))
armorK := f_clampInt(armorK, 0, 100)

int armorRes = 0
if bandOK
    armorRes += 1
    if armorState == 2 or armorState == -2
        armorRes += 2
    else if armorState == 1 or armorState == -1
        armorRes += 1

    if armorK >= kStrong
        armorRes += 2
    else if armorK >= kMid
        armorRes += 1

if isSpikeHard
    armorRes += 1
armorRes := math.max(0, math.min(5, armorRes))

int armorEvent = 0
if isSpikeHard
    armorEvent := 10
else if bandOK and armorState == 2
    armorEvent := 20
else if bandOK and armorState == -2
    armorEvent := 21
else if bandOK and (armorState == -1 or armorState == 0 or armorState == 1)
    armorEvent := 30
else
    armorEvent := 0

int armorStrongEvent = armorRes >= 4 ? 40 : 0
int armorSigDir = armorDir > 0 ? 1 : armorDir < 0 ? -1 : 0
int armorSigRes = armorRes < 0 ? 0 : armorRes > 5 ? 5 : armorRes
int armorSig = armorSigDir == 0 ? 0 : (armorSigDir * 100 + armorSigRes)
int armorActiveFlag = (armorState != 0 or armorK >= 35 or armorEvent != 0) ? 1 : 0
int armorRisk = armorK >= 70 ? 0 : armorK >= 35 ? 1 : 2

var table armorTable = na

if barstate.islast and showTable
    if na(armorTable)
        armorTable := table.new(f_tablePos(tablePosOpt), 4, 9, border_width=1)
    table.cell(armorTable, 0, 0, "KEY",  text_color=headerText, bgcolor=headerBg, text_size=f_textSize(tableSizeOpt))

    table.cell(armorTable, 0, 1, "DIR", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 1, 1, f_dirCode(armorDir), text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 2, 1, str.tostring(armorDir), text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 3, 1, "band breakout dir", text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))

    table.cell(armorTable, 0, 2, "STATE", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 1, 2, f_stateCode(armorState), text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 2, 2, str.tostring(armorState), text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 3, 2, f_note(bandOK, armorState, armorRes, armorEvent), text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))

    table.cell(armorTable, 0, 3, "SCORE", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 1, 3, armorK >= 80 ? "STR" : armorK >= 55 ? "MID" : "LOW", text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 2, 3, str.tostring(armorK), text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 3, 3, "impulse k", text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))

    table.cell(armorTable, 0, 4, "RES", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 1, 4, armorRes >= 4 ? "HARD" : armorRes >= 2 ? "MID" : "LOW", text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 2, 4, str.tostring(armorRes), text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 3, 4, "0-5", text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))

    table.cell(armorTable, 0, 5, "BAND", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 1, 5, bandOK ? "OPEN" : "OFF", text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 2, 5, bandOK ? "1" : "0", text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 3, 5, "valid impulse band", text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))

    table.cell(armorTable, 0, 6, "SPIKE", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 1, 6, isSpikeHard ? "HARD" : "NONE", text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 2, 6, isSpikeHard ? "1" : "0", text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 3, 6, "anchor reset", text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))

    table.cell(armorTable, 0, 7, "SIGNAL", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 1, 7, armorSig == 0 ? "NONE" : armorSig > 0 ? "UP_PRS" : "DN_PRS", text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 2, 7, str.tostring(armorSig), text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 3, 7, "dir*100+res", text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))

    table.cell(armorTable, 0, 8, "EVENT", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 1, 8, f_evtCode(armorEvent), text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 2, 8, str.tostring(armorEvent), text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(armorTable, 3, 8, armorStrongEvent == 40 ? "strong band" : "transition", text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
if barstate.islast and not showTable
    if not na(armorTable)
        table.delete(armorTable)
        armorTable := na
