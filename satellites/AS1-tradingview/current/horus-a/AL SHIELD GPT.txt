f_slotToDir(_slot) =>
    int dir = 0
    if _slot == "UP"
        dir := 1
    else if _slot == "DOWN"
        dir := -1
    else
        dir := 0
    dir

f_clampInt(_v, _min, _max) =>
    math.max(_min, math.min(_max, _v))

f_tablePos(_opt) =>
    pos = position.top_right
    if _opt == "TL"
        pos := position.top_left
    else if _opt == "TC"
        pos := position.top_center
    else if _opt == "TR"
        pos := position.top_right
    else if _opt == "ML"
        pos := position.middle_left
    else if _opt == "MC"
        pos := position.middle_center
    else if _opt == "MR"
        pos := position.middle_right
    else if _opt == "BL"
        pos := position.bottom_left
    else if _opt == "BC"
        pos := position.bottom_center
    else
        pos := position.bottom_right
    pos

f_textSize(_opt) =>
    sz = size.small
    if _opt == "TINY"
        sz := size.tiny
    else if _opt == "SMALL"
        sz := size.small
    else if _opt == "NORMAL"
        sz := size.normal
    else
        sz := size.large
    sz

f_dirCode(_dir) =>
    string out = "FLAT"
    if _dir > 0
        out := "UP"
    else if _dir < 0
        out := "DOWN"
    else
        out := "FLAT"
    out

f_stateCode(_state) =>
    string out = "IDLE"
    if _state == 2
        out := "UP_STR"
    else if _state == 1
        out := "UP"
    else if _state == -1
        out := "DN"
    else if _state == -2
        out := "DN_STR"
    else
        out := "IDLE"
    out

f_evtCode(_evt) =>
    string out = "NONE"
    if _evt == 11
        out := "UP_STRONG"
    else if _evt == 10
        out := "UP"
    else if _evt == 21
        out := "DOWN_STRONG"
    else if _evt == 20
        out := "DOWN"
    else if _evt == 90
        out := "FILTER_OFF"
    else
        out := "NONE"
    out

f_note(_dir, _state, _filterOK) =>
    string out = "WAIT"
    if not _filterOK
        out := "FILTER"
    else if _state == 2 and _dir > 0
        out := "HARD GATE OPEN"
    else if _state == 1 and _dir > 0
        out := "SOFT GATE OPEN"
    else if _state == -2 and _dir < 0
        out := "HARD GATE DOWN"
    else if _state == -1 and _dir < 0
        out := "SOFT GATE DOWN"
    else
        out := "NEUTRAL"
    out

        shieldState := -1
    else
        shieldState := 0

bool stateDownWeak   = shieldState == -1 and nz(shieldState[1]) != -1
