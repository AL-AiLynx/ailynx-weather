f_clamp(float x, float lo, float hi) =>
    math.max(lo, math.min(hi, x))

f_pos(string opt) =>
    p = position.top_right
    if opt == "TL"
        p := position.top_left
    else if opt == "TC"
        p := position.top_center
    else if opt == "TR"
        p := position.top_right
    else if opt == "ML"
        p := position.middle_left
    else if opt == "MC"
        p := position.middle_center
    else if opt == "MR"
        p := position.middle_right
    else if opt == "BL"
        p := position.bottom_left
    else if opt == "BC"
        p := position.bottom_center
    else
        p := position.bottom_right
    p

f_tsize(string opt) =>
    sz = size.small
    if opt == "XS"
        sz := size.tiny
    else if opt == "S"
        sz := size.small
    else if opt == "M"
        sz := size.normal
    else if opt == "L"
        sz := size.large
    else
        sz := size.huge
    sz

f_textSizeFromScale(int sc) =>
    sz = size.normal
    if sc <= 2
        sz := size.tiny
    else if sc <= 4
        sz := size.small
    else if sc <= 6
        sz := size.normal
    else if sc <= 8
        sz := size.large
    else
        sz := size.huge
    sz

f_dirCode(int d) =>
    d > 0 ? "UP" : d < 0 ? "DOWN" : "FLAT"

f_stateCode(int s) =>
    s == 4 ? "FAIL" : s == 3 ? "HOLD" : s == 2 ? "TRIG" : s == 1 ? "SET" : "IDLE"

f_riskCode(int r) =>
    r == 0 ? "SAFE" : r == 1 ? "MID" : "RISK"

f_scoreBand(float v) =>
    v >= 80 ? "EXTREME" : v >= 60 ? "HIGH" : v >= 40 ? "MID" : v >= 20 ? "LOW" : "NONE"

f_reasonCode(int rc) =>
    rc == 1 ? "FAIL" : rc == 2 ? "THIN" : rc == 3 ? "ALIGN" : rc == 4 ? "RISK" : rc == 5 ? "TRIG" : rc == 6 ? "WAIT" : "NONE"

f_numTxt(float v) =>
    str.tostring(int(math.round(v)))

f_flowChar(float x) =>
    x >= 3 ? "+" : x <= -3 ? "-" : "."

f_jsonSafe(string s) =>
    str.replace_all(str.replace_all(str.replace_all(str.tostring(s), "\\", "\\\\"), "\"", "'"), "\n", " ")

f_boolStr(bool b) =>
    b ? "true" : "false"

f_kv(string k, string v) =>
    "\"" + k + "\":\"" + f_jsonSafe(v) + "\""

f_kvNum(string k, string v) =>
    "\"" + k + "\":" + v

f_kvBool(string k, bool b) =>
    "\"" + k + "\":" + f_boolStr(b)

f_keySafe(string s) =>
    string out = f_jsonSafe(s)
    out := str.replace_all(out, ":", "_")
    out := str.replace_all(out, "/", "_")
    out := str.replace_all(out, "\\", "_")
    out := str.replace_all(out, "|", "_")
    out := str.replace_all(out, " ", "_")
    out := str.replace_all(out, ",", "_")
    out
