f_clamp01(x) =>
    math.max(0.0, math.min(1.0, x))

f_tablePos(opt) =>
    pos = position.top_right
    if opt == "TL"
        pos := position.top_left
    else if opt == "TC"
        pos := position.top_center
    else if opt == "TR"
        pos := position.top_right
    else if opt == "ML"
        pos := position.middle_left
    else if opt == "MC"
        pos := position.middle_center
    else if opt == "MR"
        pos := position.middle_right
    else if opt == "BL"
        pos := position.bottom_left
    else if opt == "BC"
        pos := position.bottom_center
    else
        pos := position.bottom_right
    pos

f_textSize(opt) =>
    sz = size.small
    if opt == "TINY"
        sz := size.tiny
    else if opt == "SMALL"
        sz := size.small
    else if opt == "NORMAL"
        sz := size.normal
    else
        sz := size.large
    sz

f_stateCode(state) =>
    out = "IDLE"
    if state == 1
        out := "SET"
    else if state == 2
        out := "TRIG"
    else if state == 3
        out := "HOLD"
    else if state == 4
        out := "FAIL"
    else
        out := "IDLE"
    out

f_scoreCode(k) =>
    out = "LOW"
    if k >= 80
        out := "STR"
    else if k >= 55
        out := "MID"
    else
        out := "LOW"
    out

f_dirCode(dir) =>
    out = "FLAT"
    if dir > 0
        out := "UP"
    else if dir < 0
        out := "DOWN"
    else
        out := "FLAT"
    out

f_evtCode(evt) =>
    out = "NONE"
    if evt == 1
        out := "ON"
    else if evt == 2
        out := "OFF"
    else if evt == 3
        out := "UPD"
    else if evt == 4
        out := "SPIKE"
    else if evt == 5
        out := "FADE"
    else if evt == 6
        out := "SOFT"
    else if evt == 7
        out := "EXHAUST"
    else if evt == 8
        out := "FAKE_PWR"
    else if evt == 9
        out := "CHASE"
    else if evt == 10
        out := "FAIL_FUEL"
    else
        out := "NONE"
    out

f_riskCode(risk) =>
    out = "SAFE"
    if risk == 2
        out := "RISK"
    else if risk == 1
        out := "MID"
    else
        out := "SAFE"
    out

f_note(state, evt, res, tfTag) =>
    out = "WAIT"
    if evt == 8
        out := "FAKE POWER"
    else if evt == 10
        out := "FAILED FUEL"
    else if evt == 9
        out := "CHASE RISK"
    else if evt == 7
        out := "FUEL EXHAUSTION"
    else if evt == 4
        out := "HARD VOL SPIKE"
    else if evt == 5
        out := "POST SPIKE FADE"
    else if evt == 6
        out := "SOFT WATCH"
    else if state == 3
        out := "FUEL COOLING"
    else if state == 1
        out := "VOLUME WATCH"
    else if res >= 4
        out := "STRONG RES"
    else
        out := tfTag
    out

int sec = timeframe.in_seconds()
float tfAdj = 1.0
if useAutoTFAdjust

    else if sec <= 14400
        tfAdj := 1.32
    else

float spikeMultEff = spikeMult * tfAdj
float fadeMultEff  = fadeMult + (tfAdj - 1.0) * 0.08
int fadeBars = fadeBarsManual
if useAutoFadeBars
    int autoBars = int(math.round(5400.0 / math.max(sec, 1)))
    fadeBars := math.max(2, autoBars)

string tfTag = "AUTO TF"
if sec <= 300
    tfTag := "TF 5M"
else if sec <= 900
    tfTag := "TF 15M"
else if sec <= 3600
    tfTag := "TF 1H"
else if sec <= 14400
    tfTag := "TF 4H"
else
    tfTag := "TF 1D+"

// [4] 기본 계산

float volBase  = ta.sma(volume, baseLen)
float volBaseN = math.max(nz(volBase, volume), 1.0)
float volRatio = volume / volBaseN
float hiPrev   = ta.highest(volume[1], spikeHiLB)
float hiPrevN  = math.max(nz(hiPrev, volume), 1.0)

float tp = hlc3
float rmf = tp * volume
float tpDelta = tp - tp[1]
float posMF = tpDelta > 0 ? rmf : 0.0
float negMF = tpDelta < 0 ? rmf : 0.0
float posMFSum = ta.sma(posMF, mfiLen) * mfiLen
float negMFSum = ta.sma(negMF, mfiLen) * mfiLen
float moneyRatio = negMFSum <= 0 ? 1000000.0 : (posMFSum / negMFSum)
float mfiValue = 100.0 - (100.0 / (1.0 + moneyRatio))

float obv = ta.cum(close > close[1] ? volume : close < close[1] ? -volume : 0.0)
float obvEma = ta.ema(obv, flowLen)
float obvGap = obv - obvEma

float hlRange = high - low
float mfm = hlRange > 0 ? (((close - low) - (high - close)) / hlRange) : 0.0
float adl = ta.cum(mfm * volume)
float adlEma = ta.ema(adl, flowLen)
float adlGap = adl - adlEma

float bodyAbs = math.abs(close - open)
float atr14 = ta.atr(14)

bool spikeByMult    = volRatio >= spikeMultEff
bool spikeByNewHigh = (not na(hiPrev)) and (volume > hiPrev)
bool spikeSoftRaw   = spikeByMult or spikeByNewHigh
bool spikeHardRaw   = spikeByMult and spikeByNewHigh

bool flowWeak = (mfiValue < 45 and obvGap < 0) or (mfiValue < 40 and adlGap < 0)
bool flowHard = (mfiValue < 35 and obvGap < 0 and adlGap < 0)

var int lastHardBar = na
bool cdOK = cooldownBars <= 0 ? true : (na(lastHardBar) or (bar_index - lastHardBar >= cooldownBars))
bool spikeHard = spikeHardRaw and cdOK and (flowHard or bodyAbs >= atr14 * 0.6)
if spikeHard
    lastHardBar := bar_index

int bsHard = ta.barssince(spikeHard)
bool inFadeWindow = (bsHard >= 0) and (bsHard <= fadeBars)
bool isFade = inFadeWindow and (volRatio <= fadeMultEff) and flowWeak
bool spikeSoft = spikeSoftRaw and (not spikeHard)

float kSpikeA = f_clamp01((volRatio - 1.0) / math.max(1e-6, spikeMultEff - 1.0))
float kSpikeB = f_clamp01(volume / hiPrevN)
float kFlowA  = f_clamp01((50.0 - mfiValue) / 30.0)
float kFlowB  = f_clamp01((-obvGap) / math.max(1e-6, math.abs(obvEma)))
float kFlowC  = f_clamp01((-adlGap) / math.max(1e-6, math.abs(adlEma)))
float kSpikeF = 100.0 * (0.45 * math.max(kSpikeA, kSpikeB) + 0.30 * kFlowA + 0.15 * kFlowB + 0.10 * kFlowC)
int kSpike = int(math.round(math.max(0.0, math.min(100.0, kSpikeF))))

float fadeDepth = f_clamp01((fadeMultEff - volRatio) / math.max(1e-6, fadeMultEff - 1.0))
float fadeFresh = 0.0
if inFadeWindow
    fadeFresh := f_clamp01(1.0 - (float(bsHard) / math.max(1.0, float(fadeBars))))
float kFadeF = 100.0 * (0.45 * fadeDepth + 0.25 * fadeFresh + 0.20 * kFlowA + 0.10 * math.max(kFlowB, kFlowC))
int kFade = int(math.round(math.max(0.0, math.min(100.0, kFadeF))))

float hlSafe = math.max(high - low, syminfo.mintick)
float closePos = f_clamp01((close - low) / hlSafe)
float bodyPart = atr14 > 0 ? f_clamp01(bodyAbs / math.max(atr14 * 0.85, syminfo.mintick)) : 0.0
float priorClose = nz(close[1], close)
float pushUpPart = f_clamp01((close - priorClose) / math.max(atr14 * 0.70, syminfo.mintick))
float pushDnPart = f_clamp01((priorClose - close) / math.max(atr14 * 0.70, syminfo.mintick))
float pushPart = math.max(pushUpPart, pushDnPart)
float closeStressPart = math.max(f_clamp01((closePos - 0.70) / 0.30), f_clamp01((0.30 - closePos) / 0.30))
float volumeDecayPart = f_clamp01((1.0 - volRatio) / 0.55)
float weakContinuationPart = pushPart > 0.25 ? f_clamp01(0.44 * volumeDecayPart + 0.36 * kFlowA + 0.20 * math.max(kFlowB, kFlowC)) : 0.0

float afterSpikeCooling = inFadeWindow ? 100.0 * f_clamp01(0.44 * fadeFresh + 0.34 * fadeDepth + 0.22 * kFlowA) : 0.0
float volumeDecayScore = 100.0 * f_clamp01(0.54 * volumeDecayPart + 0.26 * kFlowA + 0.20 * math.max(kFlowB, kFlowC))
float fuelFadeScore = math.max(float(kFade), 100.0 * f_clamp01(0.40 * volumeDecayPart + 0.30 * weakContinuationPart + 0.20 * kFlowA + 0.10 * fadeFresh))
float exhaustScore = spikeHard ? math.max(float(kSpike), 100.0 * f_clamp01(0.42 * math.max(kSpikeA, kSpikeB) + 0.25 * closeStressPart + 0.20 * bodyPart + 0.13 * kFlowA)) : 0.0
float fakePowerScore = 100.0 * f_clamp01(0.36 * weakContinuationPart + 0.26 * bodyPart + 0.22 * volumeDecayPart + 0.16 * kFlowA)
float chaseRiskScore = 100.0 * f_clamp01(0.34 * pushPart + 0.24 * closeStressPart + 0.22 * math.max(volumeDecayPart, math.max(kSpikeA, kSpikeB)) + 0.20 * kFlowA)
float failedContinuationFuelScore = inFadeWindow and pushPart < 0.18 and flowWeak ? 100.0 * f_clamp01(0.42 * fadeFresh + 0.30 * kFlowA + 0.28 * volumeDecayPart) : 0.0

float neckScoreF = math.max(math.max(exhaustScore, fakePowerScore), math.max(fuelFadeScore, math.max(volumeDecayScore, math.max(chaseRiskScore, failedContinuationFuelScore))))
int neckK = int(math.round(neckScoreF))
neckK := math.max(0, math.min(100, neckK))

bool fuelExhaustion = exhaustScore >= 65.0
bool fakePower = fakePowerScore >= 62.0 and weakContinuationPart >= 0.35
bool chaseRisk = chaseRiskScore >= 58.0
bool failedFuel = failedContinuationFuelScore >= 58.0
bool fuelFade = fuelFadeScore >= 48.0 or afterSpikeCooling >= 45.0
bool volumeWatch = spikeSoft or volumeDecayScore >= 45.0 or chaseRisk

int neckState = 0
if fakePower or failedFuel
    neckState := 4
else if fuelExhaustion or spikeHard
    neckState := 2
else if fuelFade
    neckState := 3
else if volumeWatch
    neckState := 1
else
    neckState := 0

int neckRes = 0
if fuelExhaustion or fakePower
    neckRes += 3
if failedFuel or fuelFade
    neckRes += 2
if chaseRisk and neckK >= 60
    neckRes += 1
if neckK >= 85
    neckRes += 1
else if neckK >= 65 and neckRes > 0
    neckRes += 1
neckRes := math.max(0, math.min(5, neckRes))

bool pushUp = close >= open and close >= priorClose
bool pushDn = close <= open and close <= priorClose
int neckDir = 0
if (fuelExhaustion or fakePower) and neckK >= 72
    if pushUp
        neckDir := -1
    else if pushDn
        neckDir := 1
    else
        neckDir := 0
else
    neckDir := 0

int EVT_NONE  = 0
int EVT_ON    = 1
int EVT_OFF   = 2
int EVT_UPD   = 3
int EVT_SPIKE = 4
int EVT_FADE  = 5
int EVT_SOFT  = 6
int EVT_EXHAUST = 7
int EVT_FAKE_POWER = 8
int EVT_CHASE = 9
int EVT_FAILED_FUEL = 10

int prevState = bar_index > 0 ? neckState[1] : 0
int prevK     = bar_index > 0 ? neckK[1] : 0
int prevRes   = bar_index > 0 ? neckRes[1] : 0

bool onEdge   = (prevState == 0) and (neckState != 0)
bool offEdge  = (prevState != 0) and (neckState == 0)
bool updEdge  = (neckState != 0) and (prevState != 0) and (neckK != prevK)

int neckEvt = EVT_NONE
if fakePower
    neckEvt := EVT_FAKE_POWER
else if failedFuel
    neckEvt := EVT_FAILED_FUEL
else if fuelExhaustion
    neckEvt := EVT_EXHAUST
else if chaseRisk
    neckEvt := EVT_CHASE
else if spikeHard
    neckEvt := EVT_SPIKE
else if fuelFade
    neckEvt := EVT_FADE
else if spikeSoft
    neckEvt := EVT_SOFT
else if onEdge
    neckEvt := EVT_ON
else if offEdge
    neckEvt := EVT_OFF
else if updEdge
    neckEvt := EVT_UPD
else
    neckEvt := EVT_NONE

int neckActiveFlag = (neckState != 0 or neckK >= 35 or neckEvt != 0 or volumeWatch) ? 1 : 0
int neckRisk = (neckState == 2 or neckState == 4) ? 2 : ((neckState == 3 or chaseRisk) ? 1 : 0)
