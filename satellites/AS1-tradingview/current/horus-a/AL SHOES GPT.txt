f_clamp(x, lo, hi) =>
    math.min(hi, math.max(lo, x))

f_clampInt(x, lo, hi) =>
    math.max(lo, math.min(hi, x))

f_tablePos(opt) =>
    pos = position.top_right
    if opt == "TL"
        pos := position.top_left
    else if opt == "TC"
        pos := position.top_center
    else if opt == "TR"
        pos := position.top_right
    else if opt == "ML"
        pos := position.middle_left
    else if opt == "MC"
        pos := position.middle_center
    else if opt == "MR"
        pos := position.middle_right
    else if opt == "BL"
        pos := position.bottom_left
    else if opt == "BC"
        pos := position.bottom_center
    else
        pos := position.bottom_right
    pos

f_textSize(opt) =>
    sz = size.small
    if opt == "TINY"
        sz := size.tiny
    else if opt == "SMALL"
        sz := size.small
    else if opt == "NORMAL"
        sz := size.normal
    else
        sz := size.large
    sz

f_tfScale(secNow, lo, hi) =>
    s = math.sqrt(secNow / 60.0)
    f_clamp(s, lo, hi)

f_dirCode(dir) =>
    out = "FLAT"
    if dir < 0
        out := "DOWN"
    else if dir > 0
        out := "UP"
    else
        out := "FLAT"
    out

f_stateCode(state) =>
    out = "IDLE"
    if state == 1
        out := "SET"
    else if state == 2
        out := "TRIG"
    else if state == 3
        out := "HOLD"
    else if state == 4
        out := "FAIL"
    else
        out := "IDLE"
    out

f_edgeCode(edgeSide) =>
    out = "NONE"
    if edgeSide > 0
        out := "TOP"
    else if edgeSide < 0
        out := "BOTTOM"
    else
        out := "NONE"
    out

f_scoreCode(k) =>
    out = "LOW"
    if k >= 85
        out := "STR"
    else if k >= 70
        out := "MID"
    else if k >= 35
        out := "LITE"
    else
        out := "LOW"
    out

f_evtCode(evt) =>
    out = "NONE"
    if evt == 10
        out := "TOP_WATCH"
    else if evt == 11
        out := "BOT_WATCH"
    else if evt == 20
        out := "TOP_FAIL"
    else if evt == 21
        out := "BOT_FAIL"
    else if evt == 30
        out := "CHASE"
    else if evt == 31
        out := "FATIGUE"
    else if evt == 32
        out := "REJECT"
    else if evt == 40
        out := "INVALID"
    else
        out := "NONE"
    out

f_note(tracking, active, evt, ready, edgeOK, tfTag) =>
    out = "WAIT"
    if evt == 10
        out := "TOP EDGE WATCH"
    else if evt == 11
        out := "BOTTOM EDGE WATCH"
    else if evt == 20
        out := "TOP ATTEMPT FAILED"
    else if evt == 21
        out := "BOTTOM ATTEMPT FAILED"
    else if evt == 30
        out := "CHASE RISK"
    else if evt == 31
        out := "LAST STEP FATIGUE"
    else if evt == 32
        out := "ENDPOINT REJECTION"
    else if evt == 40
        out := "EDGE INVALIDATED"
    else if active
        out := "WEAKNESS ON"
    else if tracking and edgeOK and ready
        out := "READY"
    else if tracking and edgeOK
        out := "EDGE TRACK"
    else if tracking
        out := "EDGE WATCH"
    else
        out := tfTag
    out

int secNow = timeframe.in_seconds()

float deepPullbackPctEff = f_clamp(deepPullbackPct * (1.0 - (tfAdj - 1.0) * 0.10), 0.25, 0.95)
int minUpScoreEff = f_clampInt(int(math.round(minUpScore * (1.0 + (tfAdj - 1.0) * 0.08))), 0, 100)
float adxWeakMinEff = adxWeakMin * (1.0 + (tfAdj - 1.0) * 0.15)

string tfTag = "AUTO TF"
if secNow <= 300
    tfTag := "TF 5M"
else if secNow <= 900
    tfTag := "TF 15M"
else if secNow <= 3600
    tfTag := "TF 1H"
else if secNow <= 14400
    tfTag := "TF 4H"
else
    tfTag := "TF 1D+"

float atrV = ta.atr(atrLen)
float eps = atrV * newHighEpsMult
float emaFast = ta.ema(close, emaFastLen)
float emaSlow = ta.ema(close, emaSlowLen)
float emaSlope = emaFast - nz(emaFast[1], emaFast)
float rsiV = ta.rsi(close, rsiLen)
[pdiV, mdiV, adxV] = ta.dmi(adxLen, adxSmooth)

float zoneRaw = atrV * zoneBaseMult * tfScaleVal
float zoneCap = atrV * zoneCapAtrMult
float zone = math.min(zoneRaw, zoneCap)
float highestPrev = bar_index > 0 ? ta.highest(high, highLookback)[1] : high
float lowestPrev = bar_index > 0 ? ta.lowest(low, highLookback)[1] : low

int upScore = 0
upScore += emaFast > emaSlow ? 20 : 0
upScore += close > emaFast ? 16 : 0
upScore += emaSlope > 0 ? 16 : 0
upScore += rsiV >= 60 ? 18 : rsiV >= 55 ? 9 : 0
upScore += adxV >= adxWeakMinEff ? 10 : 0
upScore += high > (highestPrev + eps) ? 20 : 0
upScore := f_clampInt(upScore, 0, 100)

int downScore = 0
downScore += emaFast < emaSlow ? 20 : 0
downScore += close < emaFast ? 16 : 0
downScore += emaSlope < 0 ? 16 : 0
downScore += rsiV <= 40 ? 18 : rsiV <= 45 ? 9 : 0
downScore += adxV >= adxWeakMinEff ? 10 : 0
downScore += low < (lowestPrev - eps) ? 20 : 0
downScore := f_clampInt(downScore, 0, 100)

bool slopeWeakTop = false
bool slopeWeakBottom = false
if slopeConfirmBars <= 1
    slopeWeakTop := emaSlope < 0
    slopeWeakBottom := emaSlope > 0
else
    bool allDown = true
    bool allUp = true
    for i = 0 to slopeConfirmBars - 1
        allDown := allDown and ((emaFast[i] - emaFast[i + 1]) < 0)
        allUp := allUp and ((emaFast[i] - emaFast[i + 1]) > 0)
    slopeWeakTop := allDown
    slopeWeakBottom := allUp

bool topSetupSignal = (upScore >= minUpScoreEff) and high > (highestPrev + eps) and close >= emaFast
bool bottomSetupSignal = (downScore >= minUpScoreEff) and low < (lowestPrev - eps) and close <= emaFast

//==================================================
// [5] 상태
//==================================================
var float anchorRefPrice = na
var int   anchorBar = na
var float edgeExtreme = na
var int   edgeBar = na
var int   edgeSide = 0
var bool  setupTracking = false
var bool  shoesTriggered = false
var bool  shoesActive = false
var int   shoesBar = na
var int   lastFireBar = na

bool gapOK = minGapAfterFire <= 0 ? true : (na(lastFireBar) or (bar_index - lastFireBar >= minGapAfterFire))

bool setupNow = false
bool firedNow = false
bool clearedNow = false
bool expiredNow = false

if (topSetupSignal or bottomSetupSignal) and gapOK and not setupTracking and not shoesActive
    int nextEdgeSide = topSetupSignal and (not bottomSetupSignal or upScore >= downScore) ? 1 : -1
    setupNow := true
    edgeSide := nextEdgeSide
    anchorRefPrice := useCloseRef ? close : (edgeSide > 0 ? high : low)
    anchorBar := bar_index
    edgeExtreme := edgeSide > 0 ? high : low
    edgeBar := bar_index
    setupTracking := true
    shoesTriggered := false
    shoesActive := false
    shoesBar := na

bool trackingAlive = setupTracking and not na(anchorBar) and ((bar_index - anchorBar) <= setupMaxBars)
if setupTracking and not trackingAlive and not shoesActive
    expiredNow := true
    setupTracking := false

if setupTracking
    if na(edgeExtreme)
        edgeExtreme := edgeSide > 0 ? high : low
        edgeBar := bar_index
    else if edgeSide > 0 and high > (edgeExtreme + eps)
        edgeExtreme := high
        edgeBar := bar_index
    else if edgeSide < 0 and low < (edgeExtreme - eps)
        edgeExtreme := low
        edgeBar := bar_index

bool topTracking = setupTracking and edgeSide > 0
bool bottomTracking = setupTracking and edgeSide < 0
bool topEdgeConfirmed = topTracking and not na(edgeExtreme) and not na(anchorRefPrice) and edgeExtreme > (anchorRefPrice + eps)
bool bottomEdgeConfirmed = bottomTracking and not na(edgeExtreme) and not na(anchorRefPrice) and edgeExtreme < (anchorRefPrice - eps)
bool edgeConfirmed = topEdgeConfirmed or bottomEdgeConfirmed
float distToAnchor = setupTracking and not na(anchorRefPrice) ? math.abs(close - anchorRefPrice) : na
bool inAnchorZone = setupTracking and not na(distToAnchor) and (distToAnchor <= zone)
bool deepPullback = setupTracking and not na(distToAnchor) and (distToAnchor <= zone * deepPullbackPctEff)
bool bearBody = close < open
bool bullBody = close > open
bool adxWeak = adxV < nz(adxV[1], adxV) and adxV <= adxWeakMinEff
float diTopGap = pdiV - mdiV
float diBottomGap = mdiV - pdiV
bool diTopWeak = topTracking and pdiV <= nz(pdiV[1], pdiV) and diTopGap <= nz(diTopGap[1], diTopGap)
bool diBottomWeak = bottomTracking and mdiV <= nz(mdiV[1], mdiV) and diBottomGap <= nz(diBottomGap[1], diBottomGap)
bool topFatigue = topTracking and (slopeWeakTop or diTopWeak or adxWeak)
bool bottomFatigue = bottomTracking and (slopeWeakBottom or diBottomWeak or adxWeak)

float candleRange = math.max(high - low, syminfo.mintick)
float closePos = f_clamp((close - low) / candleRange, 0.0, 1.0)
float upperWick = high - math.max(open, close)
float lowerWick = math.min(open, close) - low
float upperWickPart = atrV > 0 ? f_clamp(upperWick / math.max(atrV * 0.75, syminfo.mintick), 0.0, 1.0) : 0.0
float lowerWickPart = atrV > 0 ? f_clamp(lowerWick / math.max(atrV * 0.75, syminfo.mintick), 0.0, 1.0) : 0.0
float endpointTopScore = topTracking ? 100.0 * f_clamp(0.58 * upperWickPart + 0.42 * (1.0 - closePos), 0.0, 1.0) : 0.0
float endpointBottomScore = bottomTracking ? 100.0 * f_clamp(0.58 * lowerWickPart + 0.42 * closePos, 0.0, 1.0) : 0.0
float endpointRejectionScore = math.max(endpointTopScore, endpointBottomScore)

float edgeMove = setupTracking and not na(edgeExtreme) and not na(anchorRefPrice) ? math.abs(edgeExtreme - anchorRefPrice) : 0.0
float extensionPart = atrV > 0 ? f_clamp(edgeMove / math.max(atrV * 1.60, syminfo.mintick), 0.0, 1.0) : 0.0
float anchorHoldPart = zone > 0 and not na(distToAnchor) ? f_clamp(1.0 - distToAnchor / zone, 0.0, 1.0) : 0.0
float adxFadePart = f_clamp((adxWeakMinEff - adxV) / math.max(adxWeakMinEff, 1.0), 0.0, 1.0)
float slopeFatiguePart = topTracking ? f_clamp((-emaSlope) / math.max(atrV * 0.08, syminfo.mintick), 0.0, 1.0) : bottomTracking ? f_clamp(emaSlope / math.max(atrV * 0.08, syminfo.mintick), 0.0, 1.0) : 0.0
float diFatiguePart = topTracking ? f_clamp((nz(diTopGap[1], diTopGap) - diTopGap) / 12.0, 0.0, 1.0) : bottomTracking ? f_clamp((nz(diBottomGap[1], diBottomGap) - diBottomGap) / 12.0, 0.0, 1.0) : 0.0
float edgeScorePart = f_clamp(float(math.max(upScore, downScore)) / 100.0, 0.0, 1.0)

float lastStepFatigue = setupTracking ? 100.0 * f_clamp(0.34 * slopeFatiguePart + 0.26 * diFatiguePart + 0.22 * adxFadePart + 0.18 * endpointRejectionScore / 100.0, 0.0, 1.0) : 0.0
float chaseRiskScore = setupTracking ? 100.0 * f_clamp(0.34 * extensionPart + 0.26 * edgeScorePart + 0.22 * endpointRejectionScore / 100.0 + 0.18 * (topTracking ? f_clamp((rsiV - 62.0) / 18.0, 0.0, 1.0) : f_clamp((38.0 - rsiV) / 18.0, 0.0, 1.0)), 0.0, 1.0) : 0.0
float edgeWeaknessScore = setupTracking ? 100.0 * f_clamp(0.30 * anchorHoldPart + 0.25 * lastStepFatigue / 100.0 + 0.22 * endpointRejectionScore / 100.0 + 0.15 * extensionPart + 0.08 * adxFadePart, 0.0, 1.0) : 0.0

bool readyToFire = setupTracking and edgeConfirmed and inAnchorZone and deepPullback and ((topTracking and topFatigue and ((not requireBearCandle) or bearBody)) or (bottomTracking and bottomFatigue and ((not requireBearCandle) or bullBody)))
bool triggerCond = readyToFire and (not shoesTriggered) and gapOK

if triggerCond
    firedNow := true
    shoesTriggered := true
    shoesActive := true
    shoesBar := bar_index
    lastFireBar := bar_index

bool reclaimClose = false
if shoesActive and not na(anchorRefPrice)
    reclaimClose := (edgeSide > 0 and close > anchorRefPrice + eps and not topFatigue) or (edgeSide < 0 and close < anchorRefPrice - eps and not bottomFatigue)
if reclaimClose
    shoesActive := false
    setupTracking := false
    clearedNow := true

bool tooLong = shoesActive and not na(shoesBar) and ((bar_index - shoesBar) >= maxHoldBars)
if tooLong
    shoesActive := false
    setupTracking := false
    clearedNow := true

if not shoesActive and setupTracking and not expiredNow and not trackingAlive
    expiredNow := true
    setupTracking := false

bool inFade = shoesActive and not na(shoesBar) and ((bar_index - shoesBar) <= fadeBars)

//==================================================

int shoesDir = 0
int shoesState = 0
float shoesKf = 0.0

if clearedNow or expiredNow
    shoesState := 4
    shoesDir := 0
    shoesKf := math.max(edgeWeaknessScore, chaseRiskScore) * 0.45
else if shoesActive
    shoesState := firedNow ? 2 : 3
    shoesDir := edgeSide > 0 ? -1 : edgeSide < 0 ? 1 : 0
    float kRaw = math.max(edgeWeaknessScore, math.max(lastStepFatigue, endpointRejectionScore))
    if inFade
        float t = float(bar_index - shoesBar) / math.max(1.0, float(fadeBars))
        float w = 1.0 - f_clamp(t * 0.45, 0.0, 0.45)
        shoesKf := kRaw * w
    else
        shoesKf := kRaw
else if setupTracking
    shoesState := chaseRiskScore >= 62 or lastStepFatigue >= 62 or endpointRejectionScore >= 70 ? 3 : 1
    shoesDir := 0
    shoesKf := math.max(math.max(chaseRiskScore, math.max(edgeWeaknessScore * 0.82, lastStepFatigue * 0.76)), shoesState == 3 ? 45.0 : 35.0)
else
    shoesState := 0
    shoesDir := 0
    shoesKf := 0.0

int shoesK = int(math.round(f_clamp(shoesKf, 0.0, 100.0)))
int shoesRes = 0
if shoesActive
    shoesRes += 3
    shoesRes += shoesK >= 85 ? 2 : shoesK >= 70 ? 1 : 0
    shoesRes += endpointRejectionScore >= 70 ? 1 : 0
    shoesRes := f_clampInt(shoesRes, 0, 5)
else if setupTracking
    shoesRes := shoesState == 3 ? 3 : 2
else if clearedNow or expiredNow
    shoesRes := 1
else
    shoesRes := 0

int shoesEvt = 0
if setupNow
    shoesEvt := edgeSide > 0 ? 10 : 11
else if firedNow
    shoesEvt := edgeSide > 0 ? 20 : 21
else if clearedNow or expiredNow
    shoesEvt := 40
else if endpointRejectionScore >= 65
    shoesEvt := 32
else if lastStepFatigue >= 58
    shoesEvt := 31
else if chaseRiskScore >= 58
    shoesEvt := 30
else if setupTracking
    shoesEvt := edgeSide > 0 ? 10 : 11
else
    shoesEvt := 0

int shoesSigDir = shoesDir > 0 ? 1 : shoesDir < 0 ? -1 : 0
int shoesSigRes = shoesRes < 0 ? 0 : shoesRes > 5 ? 5 : shoesRes
int shoesSig = shoesSigDir == 0 ? 0 : (shoesSigDir * 100 + shoesSigRes)
int shoesActiveFlag = (setupTracking or shoesState != 0 or shoesK >= 35 or firedNow) ? 1 : 0
int shoesTrackFlag = setupTracking ? 1 : 0
int shoesRisk = (shoesState == 2 or shoesState == 3) ? 2 : shoesState == 1 ? 1 : 0

// [7] GPT 표

textSize = f_textSize(tableSizeOpt)
var table shoesTable = table.new(f_tablePos(tablePosOpt), 4, 9, border_width=1)

if barstate.islast and showTable
    table.cell(shoesTable, 0, 0, "KEY",  text_color=headerText, bgcolor=headerBg, text_size=textSize)
    table.cell(shoesTable, 1, 0, "CODE", text_color=headerText, bgcolor=headerBg, text_size=textSize)
    table.cell(shoesTable, 2, 0, "NUM",  text_color=numTextColor, bgcolor=headerBg, text_size=textSize)
    table.cell(shoesTable, 3, 0, "NOTE", text_color=headerText, bgcolor=headerBg, text_size=textSize)

    table.cell(shoesTable, 0, 1, "DIR", text_color=keyTextColor, bgcolor=cellBg, text_size=textSize)
    table.cell(shoesTable, 1, 1, f_dirCode(shoesDir), text_color=codeTextColor, bgcolor=cellBg, text_size=textSize)
    table.cell(shoesTable, 2, 1, str.tostring(shoesDir), text_color=numTextColor, bgcolor=cellBg, text_size=textSize)
    table.cell(shoesTable, 3, 1, shoesDir == 0 ? "setup flat" : "edge failure bias", text_color=noteTextColor, bgcolor=cellBg, text_size=textSize)

    table.cell(shoesTable, 0, 2, "STATE", text_color=keyTextColor, bgcolor=cellBg, text_size=textSize)
