f_clampInt(_v, _min, _max) =>
    math.max(_min, math.min(_max, _v))

f_clamp01(_v) =>
    math.max(0.0, math.min(1.0, _v))

f_tablePos(_opt) =>
    pos = position.top_right
    if _opt == "TL"
        pos := position.top_left
    else if _opt == "TC"
        pos := position.top_center
    else if _opt == "TR"
        pos := position.top_right
    else if _opt == "ML"
        pos := position.middle_left
    else if _opt == "MC"
        pos := position.middle_center
    else if _opt == "MR"
        pos := position.middle_right
    else if _opt == "BL"
        pos := position.bottom_left
    else if _opt == "BC"
        pos := position.bottom_center
    else
        pos := position.bottom_right
    pos

f_textSize(_opt) =>
    sz = size.small
    if _opt == "TINY"
        sz := size.tiny
    else if _opt == "SMALL"
        sz := size.small
    else if _opt == "NORMAL"
        sz := size.normal
    else
        sz := size.large
    sz

f_dirCode(_dir) =>
    string out = "FLAT"
    if _dir > 0
        out := "UP"
    else if _dir < 0
        out := "DOWN"
    else
        out := "FLAT"
    out

f_stateCode(_state) =>
    string out = "IDLE"
    if _state == 2
        out := "TRIG"
    else if _state == 1
        out := "SET"
    else if _state == 3
        out := "HOLD"
    else if _state == 4
        out := "FAIL"
    else
        out := "IDLE"
    out

f_scoreCode(_k) =>
    string out = "LOW"
    if _k >= 80
        out := "STR"
    else if _k >= 55
        out := "MID"
    else
        out := "LOW"
    out

f_evtCode(_evt) =>
    string out = "NONE"
    if _evt == 20
        out := "UPPER_SWEEP"
    else if _evt == 21
        out := "LOWER_SWEEP"
    else if _evt == 22
        out := "UPPER_REJECT"
    else if _evt == 23
        out := "LOWER_RECOVER"
    else if _evt == 24
        out := "SWEEP_FAIL"
    else if _evt == 25
        out := "SWEEP_ACCEPT"
    else if _evt == 26
        out := "RARE_WARN"
    else if _evt == 30
        out := "RECENT_UPPER"
    else if _evt == 31
        out := "RECENT_LOWER"
    else
        out := "NONE"
    out

f_note(_state, _evt, _res) =>
    string out = "WAIT"
    if _evt == 26
        out := "RARE SWEEP WARNING"
    else if _evt == 24
        out := "SWEEP INVALIDATED"
    else if _evt == 25
        out := "SWEEP ACCEPT"
    else if _evt == 22
        out := "UPPER SWEEP REJECT"
    else if _evt == 23
        out := "LOWER SWEEP RECOVER"
    else if _evt == 30 or _evt == 31
        out := "RECENT SWEEP EFFECT"
    else if _state == 3
        out := "POST SWEEP PRESSURE"
    else if _state == 1
        out := "SWEEP WATCH"
    else if _res >= 4
        out := "STRONG RES"
    else
        out := "NEUTRAL"
    out

// [3] 기본 계산

float atrV = ta.atr(atrLen)
bool enough = bar_index > (sweepLookback + atrLen + 10)

float wickUp = high - math.max(open, close)
float wickDn = math.min(open, close) - low
float body   = math.abs(close - open)
float candleRange = math.max(high - low, syminfo.mintick)
float closePos = f_clamp01((close - low) / candleRange)

float hiPrev = ta.highest(high[1], sweepLookback)
float loPrev = ta.lowest(low[1],  sweepLookback)

bool hiOk = not na(hiPrev)
bool loOk = not na(loPrev)

bool bodyOK = atrV > 0 ? (body >= atrV * minBodyATRMult) : true

bool rawUpperSweep = enough and hiOk and high > hiPrev and wickUp >= atrV * wickATRMult and bodyOK
bool rawLowerSweep = enough and loOk and low < loPrev and wickDn >= atrV * wickATRMult and bodyOK

bool upperSweepReject = rawUpperSweep and (close < hiPrev or (not requireRejection and closePos <= 0.45))
bool lowerSweepRecover = rawLowerSweep and (close > loPrev or (not requireRejection and closePos >= 0.55))
bool upperSweepAccept = rawUpperSweep and not upperSweepReject and close >= hiPrev
bool lowerSweepAccept = rawLowerSweep and not lowerSweepRecover and close <= loPrev

var float lastUpperRejectLevel = na
var float lastLowerRecoverLevel = na
var int lastUpperRejectBar = na
var int lastLowerRecoverBar = na

bool upperSweepFail = not upperSweepReject and not upperSweepAccept and not na(lastUpperRejectLevel) and not na(lastUpperRejectBar) and bar_index > lastUpperRejectBar and (bar_index - lastUpperRejectBar <= recentBars) and close > lastUpperRejectLevel
bool lowerSweepFail = not lowerSweepRecover and not lowerSweepAccept and not na(lastLowerRecoverLevel) and not na(lastLowerRecoverBar) and bar_index > lastLowerRecoverBar and (bar_index - lastLowerRecoverBar <= recentBars) and close < lastLowerRecoverLevel

if upperSweepReject
    lastUpperRejectLevel := hiPrev
    lastUpperRejectBar := bar_index

if lowerSweepRecover
    lastLowerRecoverLevel := loPrev
    lastLowerRecoverBar := bar_index

int barsSinceUpperReject = ta.barssince(upperSweepReject)
int barsSinceLowerRecover = ta.barssince(lowerSweepRecover)
bool recentUpperEffect = barsSinceUpperReject >= 0 and barsSinceUpperReject <= recentBars and not upperSweepFail
bool recentLowerEffect = barsSinceLowerRecover >= 0 and barsSinceLowerRecover <= recentBars and not lowerSweepFail

float upperWickPart = atrV > 0 ? f_clamp01(wickUp / math.max(atrV * wickATRMult * 1.60, syminfo.mintick)) : 0.0
float lowerWickPart = atrV > 0 ? f_clamp01(wickDn / math.max(atrV * wickATRMult * 1.60, syminfo.mintick)) : 0.0
float bodyPart = atrV > 0 ? f_clamp01(body / math.max(atrV * 0.85, syminfo.mintick)) : 0.0
float upperRejectClosePart = f_clamp01((1.0 - closePos) * 1.35)
float lowerRecoverClosePart = f_clamp01(closePos * 1.35)
float upperAcceptClosePart = f_clamp01(closePos * 1.20)
float lowerAcceptClosePart = f_clamp01((1.0 - closePos) * 1.20)

float upperSweepRejectScore = upperSweepReject ? 100.0 * f_clamp01(0.42 * upperWickPart + 0.32 * upperRejectClosePart + 0.16 * bodyPart + 0.10 * f_clamp01((high - hiPrev) / math.max(atrV, syminfo.mintick))) : 0.0
float lowerSweepRecoverScore = lowerSweepRecover ? 100.0 * f_clamp01(0.42 * lowerWickPart + 0.32 * lowerRecoverClosePart + 0.16 * bodyPart + 0.10 * f_clamp01((loPrev - low) / math.max(atrV, syminfo.mintick))) : 0.0
float sweepAcceptScore = 0.0
sweepAcceptScore := upperSweepAccept ? math.max(sweepAcceptScore, 100.0 * f_clamp01(0.44 * upperWickPart + 0.34 * upperAcceptClosePart + 0.14 * bodyPart + 0.08 * f_clamp01((close - hiPrev) / math.max(atrV, syminfo.mintick)))) : sweepAcceptScore
sweepAcceptScore := lowerSweepAccept ? math.max(sweepAcceptScore, 100.0 * f_clamp01(0.44 * lowerWickPart + 0.34 * lowerAcceptClosePart + 0.14 * bodyPart + 0.08 * f_clamp01((loPrev - close) / math.max(atrV, syminfo.mintick)))) : sweepAcceptScore

float upperRejectBase = nz(ta.valuewhen(upperSweepReject, upperSweepRejectScore, 0), 0.0)
float lowerRecoverBase = nz(ta.valuewhen(lowerSweepRecover, lowerSweepRecoverScore, 0), 0.0)
float upperRecentFade = recentUpperEffect ? 1.0 - f_clamp01(float(barsSinceUpperReject) / math.max(1.0, float(recentBars))) * 0.70 : 0.0
float lowerRecentFade = recentLowerEffect ? 1.0 - f_clamp01(float(barsSinceLowerRecover) / math.max(1.0, float(recentBars))) * 0.70 : 0.0
float recentSweepEffect = math.max(upperRejectBase * upperRecentFade, lowerRecoverBase * lowerRecentFade)

float sweepFailScore = (upperSweepFail or lowerSweepFail) ? 100.0 * f_clamp01(0.56 * recentSweepEffect / 100.0 + 0.28 * bodyPart + 0.16 * (upperSweepFail ? closePos : 1.0 - closePos)) : 0.0
float rareWarningScore = 0.0
rareWarningScore := upperSweepReject and upperSweepRejectScore >= kStrong and upperWickPart >= 0.82 and closePos <= 0.36 ? math.max(rareWarningScore, upperSweepRejectScore) : rareWarningScore
rareWarningScore := lowerSweepRecover and lowerSweepRecoverScore >= kStrong and lowerWickPart >= 0.82 and closePos >= 0.64 ? math.max(rareWarningScore, lowerSweepRecoverScore) : rareWarningScore
float sweepTrapPressure = math.max(upperSweepRejectScore, lowerSweepRecoverScore) * 0.62 + recentSweepEffect * 0.28 + sweepFailScore * 0.10
sweepTrapPressure := math.min(100.0, sweepTrapPressure)

int helmetDir = 0
if upperSweepReject and upperSweepRejectScore >= kMid
    helmetDir := -1
else if lowerSweepRecover and lowerSweepRecoverScore >= kMid
    helmetDir := 1
else
    helmetDir := 0

bool sweepWatch = rawUpperSweep or rawLowerSweep
bool confirmedSweepRisk = upperSweepReject or lowerSweepRecover or upperSweepAccept or lowerSweepAccept

int helmetState = 0
if upperSweepFail or lowerSweepFail
    helmetState := 4
else if confirmedSweepRisk or rareWarningScore > 0.0
    helmetState := 2
else if recentUpperEffect or recentLowerEffect
    helmetState := 3
else if sweepWatch
    helmetState := 1
else
    helmetState := 0

float helmetScoreF = math.max(math.max(upperSweepRejectScore, lowerSweepRecoverScore), math.max(sweepAcceptScore, math.max(sweepFailScore, math.max(recentSweepEffect, math.max(rareWarningScore, sweepTrapPressure)))))
int helmetK = int(math.round(helmetScoreF))
helmetK := f_clampInt(helmetK, 0, 100)

int helmetRes = 0
if rareWarningScore >= 75.0
    helmetRes := 5
else if math.max(upperSweepRejectScore, lowerSweepRecoverScore) >= kStrong
    helmetRes := 4
else if sweepFailScore >= 65.0 or sweepAcceptScore >= 65.0 or sweepTrapPressure >= 70.0
    helmetRes := 3
else if recentSweepEffect >= kMid
    helmetRes := 2
else if sweepWatch or helmetK >= 35
    helmetRes := 1
else
    helmetRes := 0

helmetRes := f_clampInt(helmetRes, 0, 5)

int helmetEvt = 0
if rareWarningScore > 0.0
    helmetEvt := 26
else if upperSweepFail or lowerSweepFail
    helmetEvt := 24
else if upperSweepReject
    helmetEvt := 22
else if lowerSweepRecover
    helmetEvt := 23
else if upperSweepAccept or lowerSweepAccept
    helmetEvt := 25
else if rawUpperSweep
    helmetEvt := 20
else if rawLowerSweep
    helmetEvt := 21
else if recentUpperEffect
    helmetEvt := 30
else if recentLowerEffect
    helmetEvt := 31
else
    helmetEvt := 0

int helmetActiveFlag = (helmetState != 0 or helmetK >= 35 or helmetEvt != 0) ? 1 : 0

if barstate.islast and showTable

    table.cell(helmetTable, 0, 4, "RES", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(helmetTable, 1, 4, helmetRes >= 4 ? "STRONG" : helmetRes >= 2 ? "MID" : "LOW", text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(helmetTable, 2, 4, str.tostring(helmetRes), text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(helmetTable, 3, 4, "0-5", text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))

    table.cell(helmetTable, 0, 5, "EVENT", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(helmetTable, 1, 5, f_evtCode(helmetEvt), text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(helmetTable, 2, 5, str.tostring(helmetEvt), text_color=numTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(helmetTable, 3, 5, "sweep/recent", text_color=noteTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))

    table.cell(helmetTable, 0, 6, "ACTIVE", text_color=keyTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
    table.cell(helmetTable, 1, 6, helmetActiveFlag == 1 ? "ON" : "OFF", text_color=codeTextColor, bgcolor=cellBg, text_size=f_textSize(tableSizeOpt))
