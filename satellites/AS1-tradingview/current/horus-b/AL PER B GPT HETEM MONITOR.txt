f_finalCode(int bDir, bool bLong, bool bShort, bool bWait, bool bExit, bool bBlock, int bShift) =>
    string out = "WAIT"
    if bBlock
        out := "BLOCK"
    else if bExit
        out := "RISK_FADE"
    else if bLong
        out := "UP_PRS"
    else if bShort
        out := "DN_PRS"
    else if bShift > 0
        out := "UP_FLOW"
    else if bShift < 0
        out := "DN_FLOW"
    else if bWait
        out := "WAIT"
    else if bDir > 0
        out := "UP_OBS"
    else if bDir < 0
        out := "DN_OBS"
    out

f_finalNum(string code) =>
    code == "UP_PRS" ? 1 : code == "DN_PRS" ? -1 : code == "BLOCK" ? -2 : code == "RISK_FADE" ? 2 : code == "UP_FLOW" ? 3 : code == "DN_FLOW" ? -3 : code == "UP_OBS" ? 1 : code == "DN_OBS" ? -1 : 0

f_finalNote(bool bLongOk, bool bShortOk, bool bBlock, bool bWait, int bShift, bool bBarConf) =>
    string out = bLongOk and bShortOk ? "DUAL" : bLongOk ? "UP_OK" : bShortOk ? "DN_OK" : "NONE"
    if bBlock
        out := "LOCK"
    else if bWait
        out := "WAIT"
    else if bShift != 0
        out := "SHIFT"
    else if bBarConf
        out := "CLOSE"
    else
        out := "RT"
    out
