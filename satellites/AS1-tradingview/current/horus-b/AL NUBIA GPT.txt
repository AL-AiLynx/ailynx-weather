//@version=6
indicator("AL NUBIA GPT", shorttitle="AL NUBIA G", overlay=true, max_labels_count=0, max_lines_count=0, max_bars_back=3000)

//━━━━━━━━━━━━━━━━━━
// Helpers
//━━━━━━━━━━━━━━━━━━
f_rollSum(float x, int len) =>
    float c = ta.cum(x)
    bar_index >= len ? (c - c[len]) : na

f_clamp100(float x) =>
    math.max(0.0, math.min(100.0, x))

//━━━━━━━━━━━━━━━━━━
// Constants
//━━━━━━━━━━━━━━━━━━
int DIR_DN   = -1
int DIR_FLAT = 0
int DIR_UP   = 1

int ST_IDLE = 0
int ST_SET  = 1
int ST_TRIG = 2
int ST_HOLD = 3
int ST_FAIL = 4

//━━━━━━━━━━━━━━━━━━
// Inputs
//━━━━━━━━━━━━━━━━━━
string G_CORE = "NUBIA G | CORE"
bool simpleMode = input.bool(true, "Simple Mode", group=G_CORE, display=display.none)
int profileLookback = input.int(100, "Profile Lookback", minval=20, maxval=500, group=G_CORE, display=display.none)
int profileBins = input.int(16, "Profile Bins", minval=8, maxval=40, group=G_CORE, display=display.none)
float valueAreaRatio = input.float(0.70, "Value Area Ratio", minval=0.50, maxval=0.95, step=0.01, group=G_CORE, display=display.none)
int pivotLen = input.int(10, "Pivot Length", minval=2, maxval=50, group=G_CORE, display=display.none)
int cycleLen = input.int(21, "Cycle Length", minval=5, maxval=100, group=G_CORE, display=display.none)
float regimeWeak = input.float(55.0, "Regime Weak", minval=0, maxval=100, group=G_CORE, display=display.none)
float regimeLimit = input.float(65.0, "Regime Limit", minval=0, maxval=100, group=G_CORE, display=display.none)
float regimeAggressive = input.float(75.0, "Regime Aggressive", minval=0, maxval=100, group=G_CORE, display=display.none)

string G_GATE = "NUBIA G | GATE"
bool useGate = input.bool(true, "Use Gate", group=G_GATE, display=display.none)
int gateCapR0 = input.int(0, "R0 Cap", minval=0, group=G_GATE, display=display.none)
int gateCapR1 = input.int(1, "R1 Cap", minval=0, group=G_GATE, display=display.none)
int gateCapR2 = input.int(3, "R2 Cap", minval=0, group=G_GATE, display=display.none)
int gateCapR3 = input.int(999, "R3 Cap", minval=0, group=G_GATE, display=display.none)

string G_MACRO = "NUBIA G | MACRO"
int liqSmooth = input.int(20, "Liquidity Smooth", minval=1, group=G_MACRO, display=display.none)
string ndxSymbol = input.string("NASDAQ:NDX", "Macro Symbol", group=G_MACRO, display=display.none)
int ndxEmaLen = input.int(20, "Macro EMA Length", minval=1, group=G_MACRO, display=display.none)

string G_ALERT = "NUBIA G | ALERT"
bool useAlerts = input.bool(true, "Use Alerts", group=G_ALERT, display=display.none)

string G_OUT = "NUBIA G | OUTPUT"
bool sendLinkOutputs = input.bool(true, "Send LINK5 Outputs", group=G_OUT, display=display.none)

//━━━━━━━━━━━━━━━━━━
// Engine defaults
//━━━━━━━━━━━━━━━━━━
string profileMode = "H"
int L = profileLookback
int B = profileBins
if simpleMode and timeframe.isintraday and timeframe.multiplier <= 5
    profileMode := "V"
    L := math.min(profileLookback, 80)
    B := math.min(profileBins, 12)

int lb = math.max(1, math.min(L, bar_index + 1))

//━━━━━━━━━━━━━━━━━━
// SPACE
//━━━━━━━━━━━━━━━━━━
float pocPrice = na
var float VAH = na
var float VAL = na

bool canHist = (profileMode == "H") and (lb >= 5) and (B >= 1)

if canHist
    float lo = ta.lowest(low, lb)
    float hi = ta.highest(high, lb)
    float rng = hi - lo

    if not na(lo) and not na(hi) and rng > 0
        float step = rng / float(B)
        if step > 0
            var float[] hist = array.new_float(0)
            if array.size(hist) != B
                hist := array.new_float(B, 0.0)
            else
                for i = 0 to B - 1
                    array.set(hist, i, 0.0)

            for j = 0 to lb - 1
                float p = hlc3[j]
                float v = volume[j]
                if not na(p) and not na(v)
                    int idx = int(math.floor((p - lo) / step))
                    idx := math.max(0, math.min(B - 1, idx))
                    array.set(hist, idx, array.get(hist, idx) + v)

            float maxV = -1.0
            int maxI = 0
            float totV = 0.0
            for i = 0 to B - 1
                float hv = array.get(hist, i)
                totV += hv
                if hv > maxV
                    maxV := hv
                    maxI := i

            pocPrice := lo + (float(maxI) + 0.5) * step

            if totV > 0
                float targetV = totV * valueAreaRatio
                int loI = maxI
                int hiI = maxI
                float accV = array.get(hist, maxI)

                for k = 0 to B * 2
                    if accV >= targetV
                        break
                    int nextL = loI - 1
                    int nextH = hiI + 1
                    float vL = nextL >= 0 ? array.get(hist, nextL) : -1.0
                    float vH = nextH <= B - 1 ? array.get(hist, nextH) : -1.0

                    if vH > vL
                        if nextH <= B - 1
                            hiI := nextH
                            accV += vH
                        else if nextL >= 0
                            loI := nextL
                            accV += vL
                        else
                            break
                    else
                        if nextL >= 0
                            loI := nextL
                            accV += vL
                        else if nextH <= B - 1
                            hiI := nextH
                            accV += vH
                        else
                            break

                VAL := lo + float(loI) * step
                VAH := lo + float(hiI + 1) * step

if na(pocPrice)
    float sumVol = f_rollSum(volume, lb)
    float sumPV = f_rollSum(hlc3 * volume, lb)
    if not na(sumVol) and sumVol > 0
        pocPrice := sumPV / sumVol
    VAH := na
    VAL := na

int spaceState = 0
if not na(VAH) and not na(VAL)
    spaceState := close > VAH ? 1 : close < VAL ? -1 : 0
else if not na(pocPrice)
    spaceState := close > pocPrice ? 1 : -1

//━━━━━━━━━━━━━━━━━━
// TIME
//━━━━━━━━━━━━━━━━━━
float ph = ta.pivothigh(high, pivotLen, pivotLen)
float pl = ta.pivotlow(low, pivotLen, pivotLen)

int barsSincePH = ta.barssince(not na(ph))
int barsSincePL = ta.barssince(not na(pl))

int timeState = 0
float prog = 0.0
if barsSincePL < barsSincePH
    prog := float(barsSincePL) / float(cycleLen)
    timeState := prog >= 0.8 ? -1 : 1
else
    prog := float(barsSincePH) / float(cycleLen)
    timeState := prog >= 0.8 ? 1 : -1

//━━━━━━━━━━━━━━━━━━
// SCORE + REGIME
//━━━━━━━━━━━━━━━━━━
float atr = ta.atr(14)
float valueTol = atr * 0.20
bool hasValueArea = not na(VAH) and not na(VAL)
bool hasPoc = not na(pocPrice)
bool aboveValue = hasValueArea and close > VAH
bool belowValue = hasValueArea and close < VAL
bool insideValue = hasValueArea and close <= VAH and close >= VAL
float dist = aboveValue ? (close - VAH) : belowValue ? (VAL - close) : (hasPoc ? math.abs(close - pocPrice) : 0.0)
float distAtr = atr > 0 ? dist / atr : 0.0
float spaceStrength = math.min(1.0, distAtr / 2.0)
if insideValue
    spaceStrength := math.min(spaceStrength, 0.25)

bool pocCrossUp = ta.crossover(close, pocPrice)
bool pocCrossDn = ta.crossunder(close, pocPrice)
bool pocReclaimFlag = hasPoc and (pocCrossUp or pocCrossDn)
bool pocRejectFromBelow = hasPoc and high >= pocPrice and close < pocPrice and nz(close[1], close) < pocPrice
bool pocRejectFromAbove = hasPoc and low <= pocPrice and close > pocPrice and nz(close[1], close) > pocPrice
bool pocRejectFlag = pocRejectFromBelow or pocRejectFromAbove

float timeStrength = prog < 0.8 ? (1.0 - (prog / 0.8)) : 0.10
float scoreRaw = 50.0 + (spaceState * 35.0 * spaceStrength) + (timeState * 25.0 * timeStrength)
float tsScore = f_clamp100(scoreRaw)
int rawRegimeDir = tsScore >= 55 ? DIR_UP : tsScore <= 45 ? DIR_DN : DIR_FLAT

int regime = 0
float absStr = math.abs(tsScore - 50.0) * 2.0
if absStr <= regimeWeak
    regime := 0
else if absStr <= regimeLimit
    regime := 1
else if absStr < regimeAggressive
    regime := 2
else
    regime := 3

bool valueHoldUp = rawRegimeDir == DIR_UP and ((hasValueArea and close >= VAL - valueTol and ((hasPoc and close >= pocPrice) or aboveValue)) or (not hasValueArea and hasPoc and close >= pocPrice - valueTol))
bool valueHoldDn = rawRegimeDir == DIR_DN and ((hasValueArea and close <= VAH + valueTol and ((hasPoc and close <= pocPrice) or belowValue)) or (not hasValueArea and hasPoc and close <= pocPrice + valueTol))
bool valueHoldFlag = valueHoldUp or valueHoldDn or (rawRegimeDir == DIR_FLAT and insideValue)
bool valueFailUp = rawRegimeDir == DIR_UP and ((hasValueArea and close < VAL - valueTol) or (not hasValueArea and hasPoc and close < pocPrice - valueTol))
bool valueFailDn = rawRegimeDir == DIR_DN and ((hasValueArea and close > VAH + valueTol) or (not hasValueArea and hasPoc and close > pocPrice + valueTol))
bool valueFailFlag = valueFailUp or valueFailDn

//━━━━━━━━━━━━━━━━━━
// MACRO
//━━━━━━━━━━━━━━━━━━
float fed = request.security("FRED:WALCL", "D", close)
float tga = request.security("FRED:WTREGEN", "D", close)
float rrp = request.security("FRED:RRPONTSYD", "D", close)
float rawLiq = fed - tga - rrp
float liq = ta.ema(rawLiq, liqSmooth)
float liqChg = liq - liq[1]

float ndxClose = request.security(ndxSymbol, "D", close)
float ndxEma = request.security(ndxSymbol, "D", ta.ema(close, ndxEmaLen))
bool isBull = not na(ndxClose) and not na(ndxEma) ? ndxClose >= ndxEma : false

bool macRiskOn = (liqChg > 0) and isBull
bool macRiskOff = (liqChg < 0) and (not isBull)
bool macConflict = (tsScore >= 55 and macRiskOff) or (tsScore <= 45 and macRiskOn)

int regimeAdj = regime
if macConflict
    regimeAdj := math.max(0, regime - 1)

//━━━━━━━━━━━━━━━━━━
// Gate
//━━━━━━━━━━━━━━━━━━
int capToday = regimeAdj == 0 ? gateCapR0 : regimeAdj == 1 ? gateCapR1 : regimeAdj == 2 ? gateCapR2 : gateCapR3
int dayKey = int(time("D"))
var int dayKeyPrev = na
var int gateCnt = 0

if na(dayKeyPrev)
    dayKeyPrev := dayKey
if dayKey != dayKeyPrev
    gateCnt := 0
    dayKeyPrev := dayKey

float upThr = regimeAdj == 1 ? 65 : regimeAdj == 2 ? 60 : 55
float dnThr = regimeAdj == 1 ? 35 : regimeAdj == 2 ? 40 : 45
bool gateTrigBase = (regimeAdj > 0) and (tsScore >= upThr or tsScore <= dnThr)
bool gateEdge = gateTrigBase and not gateTrigBase[1]
bool gateAllow = (not useGate) ? true : (gateCnt < capToday)
bool gateTrig = useGate ? (gateEdge and gateAllow) : gateEdge

if barstate.isconfirmed and gateTrig
    gateCnt += 1

bool regimeChanged = barstate.isconfirmed and (regimeAdj != regimeAdj[1])

//━━━━━━━━━━━━━━━━━━
// Core state
//━━━━━━━━━━━━━━━━━━
float nubiaRegimeScore = f_clamp100(absStr * 0.55 + float(regimeAdj) * 16.0 + (valueHoldFlag ? 12.0 : 0.0) + (gateTrigBase ? 8.0 : 0.0) - (macConflict ? 24.0 : 0.0))
float nubiaValuePositionScore = 35.0
if valueFailFlag
    nubiaValuePositionScore := 18.0
else if pocReclaimFlag or valueHoldFlag
    nubiaValuePositionScore := 72.0 + spaceStrength * 18.0
else if aboveValue or belowValue
    nubiaValuePositionScore := 62.0 + spaceStrength * 24.0
else if insideValue
    nubiaValuePositionScore := 48.0 + spaceStrength * 20.0
else if hasPoc
    nubiaValuePositionScore := 50.0 + spaceStrength * 20.0
nubiaValuePositionScore := f_clamp100(nubiaValuePositionScore)

float nubiaMacroConflictScore = macConflict ? f_clamp100(68.0 + float(regime) * 8.0 + (valueFailFlag ? 8.0 : 0.0)) : 0.0
float nubiaGateTriggerScore = gateTrig ? 100.0 : gateTrigBase ? 62.0 : not gateAllow ? 38.0 : 0.0
float nubiaNeutralWaitScore = (not macConflict and not valueFailFlag and regimeAdj == 0) ? f_clamp100(86.0 - math.abs(tsScore - 50.0) * 4.0 + (insideValue ? 10.0 : 0.0)) : 0.0
float nubiaTimeScore = f_clamp100(20.0 + (timeStrength * 60.0))
float nubiaMacroClarityScore = macConflict ? 24.0 : macRiskOn or macRiskOff ? 70.0 : 50.0
float nubiaScore = f_clamp100(nubiaRegimeScore * 0.34 + nubiaValuePositionScore * 0.26 + nubiaTimeScore * 0.14 + nubiaMacroClarityScore * 0.16 + nubiaGateTriggerScore * 0.10)
if nubiaNeutralWaitScore >= 60
    nubiaScore := math.max(nubiaScore, 42.0)
if macConflict or valueFailFlag
    nubiaScore := math.min(nubiaScore, 44.0)

bool regimeSupportUp = not macConflict and not valueFailFlag and regimeAdj >= 1 and rawRegimeDir == DIR_UP and (valueHoldUp or pocCrossUp or aboveValue or (hasPoc and close >= pocPrice))
bool regimeSupportDn = not macConflict and not valueFailFlag and regimeAdj >= 1 and rawRegimeDir == DIR_DN and (valueHoldDn or pocCrossDn or belowValue or (hasPoc and close <= pocPrice))
int desiredDir = regimeSupportUp ? DIR_UP : regimeSupportDn ? DIR_DN : DIR_FLAT
var int nubiaDirMemory = DIR_FLAT
bool keepUp = nubiaDirMemory == DIR_UP and not macConflict and not valueFailFlag and tsScore >= 50 and (valueHoldUp or aboveValue or (hasPoc and close >= pocPrice))
bool keepDn = nubiaDirMemory == DIR_DN and not macConflict and not valueFailFlag and tsScore <= 50 and (valueHoldDn or belowValue or (hasPoc and close <= pocPrice))
int nubiaDir = desiredDir != DIR_FLAT ? desiredDir : keepUp ? DIR_UP : keepDn ? DIR_DN : DIR_FLAT
nubiaDirMemory := nubiaDir

bool regimeHoldFlag = not macConflict and not valueFailFlag and regimeAdj >= 2 and gateAllow and (valueHoldFlag or gateTrigBase)
bool neutralWait = nubiaNeutralWaitScore >= 62 and nubiaDir == DIR_FLAT
int nubiaState = ST_IDLE
if macConflict or valueFailFlag
    nubiaState := ST_FAIL
else if gateTrig or (regimeChanged and regimeAdj > regimeAdj[1])
    nubiaState := ST_TRIG
else if regimeHoldFlag
    nubiaState := ST_HOLD
else if regimeAdj == 1 or gateTrigBase
    nubiaState := ST_SET
else
    nubiaState := ST_IDLE

//━━━━━━━━━━━━━━━━━━
// Standardized B-group outputs
//━━━━━━━━━━━━━━━━━━
bool prevMacConflict = bar_index > 0 ? macConflict[1] : false
bool prevValueFail = bar_index > 0 ? valueFailFlag[1] : false
bool prevNeutralWait = bar_index > 0 ? neutralWait[1] : false
int prevRegimeAdj = bar_index > 0 ? regimeAdj[1] : regimeAdj

int nubiaEvt = 0
if macConflict and not prevMacConflict
    nubiaEvt := 51
else if valueFailFlag and not prevValueFail
    nubiaEvt := 42
else if pocReclaimFlag
    nubiaEvt := 31
else if pocRejectFlag
    nubiaEvt := 32
else if gateTrig and nubiaDir > 0
    nubiaEvt := 11
else if gateTrig and nubiaDir < 0
    nubiaEvt := 12
else if regimeChanged and regimeAdj > prevRegimeAdj
    nubiaEvt := 21
else if regimeChanged and regimeAdj < prevRegimeAdj
    nubiaEvt := 22
else if neutralWait and not prevNeutralWait
    nubiaEvt := 61
else
    nubiaEvt := 0

int nubiaActiveFlag = (nubiaState != ST_IDLE or regimeAdj > 0 or gateTrigBase or macConflict or valueHoldFlag or valueFailFlag or neutralWait) ? 1 : 0

//━━━━━━━━━━━━━━━━━━
// Alerts
//━━━━━━━━━━━━━━━━━━
alertcondition(useAlerts and gateTrig, title="AL NUBIA G | TRIGGER", message="AL_NUBIA_G,TRIGGER")
alertcondition(useAlerts and macConflict and not prevMacConflict, title="AL NUBIA G | CONFLICT", message="AL_NUBIA_G,CONFLICT")
alertcondition(useAlerts and regimeChanged and regimeAdj > prevRegimeAdj, title="AL NUBIA G | REGIME UP", message="AL_NUBIA_G,REGIME_UP")
alertcondition(useAlerts and regimeChanged and regimeAdj < prevRegimeAdj, title="AL NUBIA G | REGIME DOWN", message="AL_NUBIA_G,REGIME_DOWN")
alertcondition(useAlerts and pocReclaimFlag, title="AL NUBIA G | POC RECLAIM", message="AL_NUBIA_G,POC_RECLAIM")
alertcondition(useAlerts and pocRejectFlag, title="AL NUBIA G | POC REJECT", message="AL_NUBIA_G,POC_REJECT")
alertcondition(useAlerts and valueFailFlag and not prevValueFail, title="AL NUBIA G | VALUE FAIL", message="AL_NUBIA_G,VALUE_FAIL")

//━━━━━━━━━━━━━━━━━━
// Hidden outputs (B-group LINK5)
//━━━━━━━━━━━━━━━━━━
plot(sendLinkOutputs ? float(nubiaDir) : na,        title="AL_NUBIA_DIRECTION",   display=display.none)
plot(sendLinkOutputs ? float(nubiaState) : na,      title="AL_NUBIA_STATE",       display=display.none)
plot(sendLinkOutputs ? nubiaScore : na,             title="AL_NUBIA_SCORE",       display=display.none)
plot(sendLinkOutputs ? float(nubiaEvt) : na,        title="AL_NUBIA_EVENT",       display=display.none)
plot(sendLinkOutputs ? float(nubiaActiveFlag) : na, title="AL_NUBIA_ACTIVE_FLAG", display=display.none)
plot(nubiaMacroConflictScore, title="AL_NUBIA_MACRO_CONFLICT_SCORE", display=display.none)
plot(nubiaNeutralWaitScore, title="AL_NUBIA_NEUTRAL_WAIT_SCORE", display=display.none)
