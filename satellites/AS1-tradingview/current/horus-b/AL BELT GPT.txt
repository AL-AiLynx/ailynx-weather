f_tablePos(_opt) =>
    pos = position.top_right
    if _opt == "TL"
        pos := position.top_left
    else if _opt == "TC"
        pos := position.top_center
    else if _opt == "TR"
        pos := position.top_right
    else if _opt == "ML"
        pos := position.middle_left
    else if _opt == "MC"
        pos := position.middle_center
    else if _opt == "MR"
        pos := position.middle_right
    else if _opt == "BL"
        pos := position.bottom_left
    else if _opt == "BC"
        pos := position.bottom_center
    else
        pos := position.bottom_right
    pos

f_textSize(_opt) =>
    sz = size.small
    if _opt == "TINY"
        sz := size.tiny
    else if _opt == "SMALL"
        sz := size.small
    else if _opt == "NORMAL"
        sz := size.normal
    else
        sz := size.large
    sz

f_dirCode(_dir) =>
    string out = "FLAT"
    if _dir > 0
        out := "LONG"
    else if _dir < 0
        out := "SHORT"
    out

f_stateCode(_state) =>
    string out = "IDLE"
    if _state == -1
        out := "SQUEEZE"
    else if _state == 2
        out := "ACCEPT"
    else if _state == 1
        out := "TRACK"
    out

f_scoreCode(_k) =>
    string out = "LOW"
    if _k >= 80
        out := "STR"
    else if _k >= 55
        out := "MID"
    out

f_evtCode(_evt) =>
    string out = "NONE"
    if _evt == 10
        out := "SQZ_ON"
    else if _evt == 11
        out := "RELEASE"
    else if _evt == 20
        out := "EXP_ST"
    else if _evt == 21
        out := "EXP_END"
    else if _evt == 30
        out := "R1"
    else if _evt == 31
        out := "R2"
    else if _evt == 39
        out := "TRAP"
    out

f_riskCode(_risk) =>
    string out = "MID"
    if _risk == 0
        out := "SAFE"
    else if _risk == 2
        out := "RISK"
    out

f_note(_state, _evt, _res, _comp, _exp) =>
    string out = "WAIT"
    if _evt == 39
        out := "TRAP WATCH"
    else if _evt == 31
        out := "DIR MATCH"
    else if _evt == 30
        out := "COMP READY"
    else if _evt == 20
        out := "EXPANSION"
    else if _evt == 11
        out := "RELEASE"
    else if _state == -1 and _comp >= 70
        out := "TIGHT SQUEEZE"
    else if _state == 1 and _exp >= 60
        out := "ACTIVE PUSH"
    else if _res >= 4
        out := "STRONG RES"
    out

float atrV = ta.atr(atrLen)
float bbBasis = ta.sma(close, bbLen)
float bbDev   = bbMult * ta.stdev(close, bbLen)
float bbUp    = bbBasis + bbDev
float bbDn    = bbBasis - bbDev
float kcBasis = ta.ema(close, kcLen)
float kcRange = kcMult * atrV
float kcUp    = kcBasis + kcRange
float kcDn    = kcBasis - kcRange
bool sqzOn  = (bbUp <= kcUp) and (bbDn >= kcDn)
bool sqzOff = not sqzOn
float bbW = math.max(bbUp - bbDn, 1e-6)
float kcW = math.max(kcUp - kcDn, 1e-6)
float wRel = bbW / kcW

float wMin  = ta.lowest(wRel, compLookback)
float wMax  = ta.highest(wRel, compLookback)
float wSpan = math.max(wMax - wMin, 1e-6)
float comp01 = 1.0 - ((wRel - wMin) / wSpan)
comp01 := math.max(0.0, math.min(1.0, comp01))
int compPct = int(math.round(100.0 * comp01))

bool prevSqzOn = bar_index > 0 ? sqzOn[1] : false
bool release = prevSqzOn and sqzOff
float prevClose = bar_index > 0 ? close[1] : close
float bbWPrev = bar_index > 0 ? bbW[1] : bbW
float grow    = (bbW - bbWPrev) / math.max(bbWPrev, 1e-6)
float moveAbs = math.abs(close - prevClose)
bool  expBar  = (grow >= expNeedGrow) and (moveAbs >= atrV * expNeedMoveA)

bool releaseConfirm = true
if requireExpConfirm
    releaseConfirm := expBar or (moveAbs >= atrV * (expNeedMoveA * 1.20))
bool releaseUnconfirmed = release and not releaseConfirm

var bool expActive   = false
var int  expStartBar = na
var int  weakCount   = 0
if release and releaseConfirm
    expActive := true
    expStartBar := bar_index
    weakCount := 0
if expActive
    if expBar
        weakCount := 0
    else
        weakCount += 1
else
    weakCount := 0

bool expTooLong   = expActive and not na(expStartBar) and ((bar_index - expStartBar) >= expMaxBars)
bool expMinHoldOK = (not expActive) or na(expStartBar) ? true : ((bar_index - expStartBar) >= minHoldBars)
bool expFizzle    = expActive and expMinHoldOK and (weakCount >= fizzleNeedBars)
bool expEndedNow = false
if expTooLong or expFizzle
    expActive := false
    expStartBar := na
    weakCount := 0
    expEndedNow := true

int expPct = 0
if expActive and not na(expStartBar)
    float t = float(bar_index - expStartBar)
    float span = math.max(1.0, float(expMaxBars))
    float time01 = math.max(0.0, math.min(1.0, 1.0 - (t / span)))
    float grow01 = math.max(0.0, math.min(1.0, grow / math.max(expNeedGrow, 1e-6)))
    float move01 = math.max(0.0, math.min(1.0, moveAbs / math.max(atrV * expNeedMoveA, 1e-6)))
    float s01 = math.max(grow01, move01)
    expPct := int(math.round(100.0 * (0.65 * s01 + 0.35 * time01)))
else
    expPct := 0

int barsSinceSqzOff = int(nz(ta.barssince(not sqzOn), 0))
int sqzBars = sqzOn ? barsSinceSqzOff : 0
bool dragState = sqzOn and compPct >= 70 and sqzBars >= minHoldBars
float kcSlope = kcBasis - (bar_index > 0 ? kcBasis[1] : kcBasis)
bool biasUp = close >= kcBasis and kcSlope > 0
bool biasDn = close <= kcBasis and kcSlope < 0
int structureDir = biasUp ? 1 : biasDn ? -1 : 0
int priceConfirmDir = moveAbs >= atrV * expNeedMoveA ? (close > prevClose ? 1 : close < prevClose ? -1 : 0) : 0
int releaseDir = priceConfirmDir != 0 ? priceConfirmDir : expActive and expPct >= 45 ? structureDir : 0
bool expansionAcceptBase = expActive and releaseDir != 0 and (expBar or expPct >= 60) and not releaseUnconfirmed
int beltTrackState = sqzOn ? -1 : expActive ? 1 : 0

bool brLinked    = (srcBraceDir != close) or (srcBraceK != close) or (srcBraceEvent != close)
bool watchLinked = (srcWatchAxis != close)
int brDir     = brLinked ? int(math.round(nz(srcBraceDir, 0.0))) : 0
int brK       = brLinked ? int(math.round(nz(srcBraceK, 0.0))) : 0
int brEvent   = brLinked ? int(math.round(nz(srcBraceEvent, 0.0))) : 0
int watchAxis = watchLinked ? int(math.round(nz(srcWatchAxis, 0.0))) : 0
brDir := brDir > 1 ? 1 : brDir < -1 ? -1 : brDir
brK   := math.max(0, math.min(100, brK))

bool watchArmed = (not useWatchTight) or (not watchLinked) or (watchAxis > 0)
bool R1 = watchArmed and (beltTrackState == -1) and (compPct >= 70) and (expPct <= 10) and (brEvent != 0 or brK >= 60)
bool dirMatch = (brDir != 0) and (releaseDir != 0) and (brDir == releaseDir)
bool R2raw = watchArmed and expansionAcceptBase and (not brLinked or brDir == 0 or dirMatch)
var int r2Streak = 0
int prevR2Streak = bar_index > 0 ? r2Streak[1] : 0
r2Streak := R2raw ? (prevR2Streak + 1) : 0
bool R2 = r2Streak >= r2ConfirmBars
bool lowEnergy = (beltTrackState == 0) and (compPct <= 25) and (expPct <= 20)
bool phaseBad  = (beltTrackState == -1 and expPct > 40) or (beltTrackState == 1 and compPct > 70)
bool brDirConflict = brDir != 0 and releaseDir != 0 and brDir != releaseDir
bool R3Trap    = (brEvent != 0 or brK >= 60) and (lowEnergy or phaseBad or brDirConflict)
bool expansionFail = expFizzle or releaseUnconfirmed or (expActive and weakCount >= fizzleNeedBars and expPct < 35)
bool fakeRelease = R3Trap or expansionFail
bool expansionAccept = expansionAcceptBase and not fakeRelease and (R2 or not brLinked or brK < 60)
int beltDir = expansionAccept ? releaseDir : 0
int beltState = sqzOn ? -1 : expansionAccept ? 2 : expActive ? 1 : 0
float compressionScore = sqzOn ? float(compPct) : 0.0
float releaseScore = (release and releaseConfirm) or expActive ? float(math.max(expPct, release ? 45 : 0)) : 0.0
float expansionAcceptScore = expansionAccept ? math.max(releaseScore, 60.0) : 0.0
float fakeReleaseScore = fakeRelease ? math.max(releaseScore, releaseUnconfirmed ? 55.0 : expFizzle ? 50.0 : 0.0) : 0.0
float trapPressureScore = R3Trap ? float(math.max(60, brK)) : expansionFail ? math.max(fakeReleaseScore, 45.0) : 0.0
float beltRawK = fakeRelease ? math.max(fakeReleaseScore, trapPressureScore) : expansionAccept ? expansionAcceptScore : beltState == -1 ? float(compPct) : beltState == 1 ? releaseScore : 0.0
int beltK = int(math.round(math.max(0.0, math.min(100.0, beltRawK))))

int beltRes = 0
if R1
    beltRes += 2
if R2
    beltRes += 3
if brEvent != 0
    beltRes += 1
if R3Trap
    beltRes -= 2
beltRes := math.max(0, math.min(5, beltRes))

bool prevExpActive = bar_index > 0 ? expActive[1] : false
bool sqzOnStart = sqzOn and not prevSqzOn
bool sqzOffRel  = release
bool expStart   = expActive and not prevExpActive
bool expEnd     = expEndedNow or ((not expActive) and prevExpActive)

int beltEvt = 0
if sqzOnStart
    beltEvt := 10
else if fakeRelease

else if expEnd

else if R1

else if R2
    beltEvt := 31
else
    beltEvt := 0

int beltActiveFlag = (beltState != 0 or beltK >= 35 or beltEvt != 0 or fakeRelease or dragState) ? 1 : 0
int beltRisk = fakeRelease or trapPressureScore >= 60 ? 2 : expansionAccept or beltRes >= 4 ? 0 : 1
