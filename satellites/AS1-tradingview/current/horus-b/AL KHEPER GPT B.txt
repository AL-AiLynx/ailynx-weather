//==================================================
// Helpers
//==================================================
f_clamp(float x, float lo, float hi) =>
    math.max(lo, math.min(hi, x))

f_clamp100(float x) =>
    f_clamp(x, 0.0, 100.0)

f_signDir(float x, float deadband) =>
    int out = DIR_FLAT
    if x > deadband
        out := DIR_UP
    else if x < -deadband
        out := DIR_DN
    out

f_autoNextTf() =>
    int sec = timeframe.in_seconds(timeframe.period)
    string out = "D"
    if sec <= 60
        out := "5"
    else if sec <= 300
        out := "15"
    else if sec <= 900
        out := "60"
    else if sec <= 3600
        out := "240"
    else if sec <= 14400
        out := "D"
    else if sec <= 86400
        out := "W"
    else if sec <= 604800
        out := "M"
    else
        out := "12M"
    out

f_tablePos(string opt) =>
    pos = position.top_right
    if opt == "TL"
        pos := position.top_left
    else if opt == "TC"
        pos := position.top_center
    else if opt == "TR"
        pos := position.top_right
    else if opt == "ML"
        pos := position.middle_left
    else if opt == "MC"
        pos := position.middle_center
    else if opt == "MR"
        pos := position.middle_right
    else if opt == "BL"
        pos := position.bottom_left
    else if opt == "BC"
        pos := position.bottom_center
    else
        pos := position.bottom_right
    pos

f_textSize(int sc) =>
    txtSz = size.normal
    if sc <= 2
        txtSz := size.tiny
    else if sc <= 4
        txtSz := size.small
    else if sc <= 6
        txtSz := size.normal
    else if sc <= 8
        txtSz := size.large
    else
        txtSz := size.huge
    txtSz

f_numTxt(float v) =>
    str.tostring(math.round(nz(v, 0.0) * 10.0) / 10.0)

f_dirText(int d) =>
    string out = "FLAT"
    if d > 0
        out := "UP"
    else if d < 0
        out := "DN"
    out

f_stateText(int s) =>
    string out = "IDLE"

        out := "APPROACH"
    else if s == ST_TEST
        out := "TEST"
    else if s == ST_HANDOFF

    else if s == ST_BLOCK
        out := "BLOCK"

    else if ec == EC_UPPER_GATE

    else if ec == EC_LOWER_GATE

        out := "CLEARING"
    else if ec == EC_ACCEL
        out := "ACCEL"
    else if ec == EC_STALL

    string safe = str.replace_all(v, "\\", "\\\\")
    safe := str.replace_all(safe, "\"", "\\\"")
    "\"" + k + "\":\"" + safe + "\""

f_jsonNumPair(string k, string v) =>
    "\"" + k + "\":" + v

//==================================================
// HTF gate model
//==================================================
string nextTf = useAutoHtf ? f_autoNextTf() : manualHtf

float htfHighRaw = request.security(syminfo.tickerid, nextTf, ta.highest(high, htfGateLen)[1], barmerge.gaps_off, barmerge.lookahead_off)
float htfLowRaw = request.security(syminfo.tickerid, nextTf, ta.lowest(low, htfGateLen)[1], barmerge.gaps_off, barmerge.lookahead_off)
float htfBasisRaw = request.security(syminfo.tickerid, nextTf, ta.ema(close, htfLen)[1], barmerge.gaps_off, barmerge.lookahead_off)
float htfBasisPrevRaw = request.security(syminfo.tickerid, nextTf, ta.ema(close, htfLen)[2], barmerge.gaps_off, barmerge.lookahead_off)
float htfAtrRaw = request.security(syminfo.tickerid, nextTf, ta.atr(atrLen)[1], barmerge.gaps_off, barmerge.lookahead_off)
float htfCloseLive = request.security(syminfo.tickerid, nextTf, close, barmerge.gaps_off, barmerge.lookahead_off)

float htfHigh = nz(htfHighRaw, high)
float htfLow = nz(htfLowRaw, low)
float htfBasis = nz(htfBasisRaw, close)
float htfBasisPrev = nz(htfBasisPrevRaw, htfBasis)
float htfAtr = math.max(nz(htfAtrRaw, ta.atr(atrLen)), syminfo.mintick)
float htfRange = math.max(htfHigh - htfLow, syminfo.mintick)
float htfMid = (htfHigh + htfLow) * 0.5
float gateNearDist = htfAtr * gateNearAtr
float gateBreakDist = htfAtr * gateBreakAtr

float htfSlopeAtr = htfAtr > 0.0 ? (htfBasis - htfBasisPrev) / htfAtr : 0.0
int htfTrendDir = DIR_FLAT
if htfCloseLive > htfBasis and htfSlopeAtr >= -htfSlopeMin
    htfTrendDir := DIR_UP
else if htfCloseLive < htfBasis and htfSlopeAtr <= htfSlopeMin
    htfTrendDir := DIR_DN

bool aboveUpperGate = close > htfHigh + gateBreakDist
bool belowLowerGate = close < htfLow - gateBreakDist
bool nearUpperGate = not aboveUpperGate and math.abs(close - htfHigh) <= gateNearDist
bool nearLowerGate = not belowLowerGate and math.abs(close - htfLow) <= gateNearDist
bool nearAnyGate = nearUpperGate or nearLowerGate or aboveUpperGate or belowLowerGate

int htfPosition = 0
if aboveUpperGate
    htfPosition := 3
else if nearUpperGate
    htfPosition := 2
else if close > htfMid
    htfPosition := 1
else if belowLowerGate
    htfPosition := -3
else if nearLowerGate
    htfPosition := -2
else if close < htfMid
    htfPosition := -1

float midDistanceN = f_clamp(math.abs(close - htfMid) / (htfRange * 0.5), 0.0, 1.0)
float gatePressureScore = nearAnyGate ? 68.0 : midDistanceN * 42.0
if aboveUpperGate or belowLowerGate
    gatePressureScore := 84.0

//==================================================
// Current TF impulse and velocity
//==================================================
float atrNow = math.max(ta.atr(atrLen), syminfo.mintick)
float fastMa = ta.ema(close, fastLen)
float slowMa = ta.ema(close, slowLen)
float impulseRaw = close - nz(close[impulseLen], close)
float impulseAtr = impulseRaw / atrNow
float spreadAtr = (fastMa - slowMa) / atrNow
float slopeAtr = (fastMa - nz(fastMa[1], fastMa)) / atrNow

int impulseDir = f_signDir(impulseAtr, impulseAtrMin)
int spreadDir = f_signDir(spreadAtr, impulseAtrMin * 0.45)
int slopeDir = f_signDir(slopeAtr, impulseAtrMin * 0.20)

int moveDir = DIR_FLAT
if impulseDir != DIR_FLAT
    moveDir := impulseDir
else if spreadDir != DIR_FLAT
    moveDir := spreadDir
else if slopeDir != DIR_FLAT
    moveDir := slopeDir

float structureVelocityRaw = ta.ema(spreadAtr + slopeAtr * 2.0, smoothLen)
float velocityAbs = math.abs(structureVelocityRaw)
float prevVelocityAbs = math.abs(nz(structureVelocityRaw[1], structureVelocityRaw))
float velocityDelta = velocityAbs - prevVelocityAbs

int kheperVelocity = 0
if velocityAbs <= velocityStallMax
    kheperVelocity := -2
else if velocityDelta > velocityDeltaMin and velocityAbs >= velocityMoveMin
    kheperVelocity := 2
else if velocityDelta < -velocityDeltaMin and prevVelocityAbs >= velocityMoveMin
    kheperVelocity := -1
else if velocityAbs >= velocityMoveMin
    kheperVelocity := 1

//==================================================
// Handoff and bridge
//==================================================
bool upProbe = moveDir > 0 and (nearUpperGate or aboveUpperGate or close > htfBasis)
bool dnProbe = moveDir < 0 and (nearLowerGate or belowLowerGate or close < htfBasis)
bool upperReject = upProbe and nearUpperGate and close < htfHigh and close < nz(close[1], close)
bool lowerReject = dnProbe and nearLowerGate and close > htfLow and close > nz(close[1], close)
bool htfSupportsUp = htfTrendDir >= 0 and htfCloseLive >= htfBasis
bool htfSupportsDn = htfTrendDir <= 0 and htfCloseLive <= htfBasis
bool upAccepted = upProbe and not upperReject and htfSupportsUp and (aboveUpperGate or (close > htfBasis and htfSlopeAtr >= -htfSlopeMin))
bool dnAccepted = dnProbe and not lowerReject and htfSupportsDn and (belowLowerGate or (close < htfBasis and htfSlopeAtr <= htfSlopeMin))
bool htfOpposesMove = moveDir != DIR_FLAT and htfTrendDir != DIR_FLAT and moveDir != htfTrendDir
bool bridgeConflict = nearAnyGate and htfOpposesMove
bool bridgeCongested = nearAnyGate and (kheperVelocity == -2 or kheperVelocity == -1 or htfOpposesMove)
bool bridgeClearing = (upAccepted or dnAccepted) or (nearAnyGate and moveDir == htfTrendDir and kheperVelocity >= 1)
bool bridgeBlocked = upperReject or lowerReject or bridgeConflict

int kheperHandoff = 0
if upAccepted
    kheperHandoff := 2
else if dnAccepted
    kheperHandoff := -2
else if upperReject
    kheperHandoff := 3
else if lowerReject
    kheperHandoff := -3
else if upProbe
    kheperHandoff := 1
else if dnProbe
    kheperHandoff := -1

int kheperBridge = BR_OPEN
if bridgeBlocked
    kheperBridge := BR_BLOCKED
else if bridgeCongested
    kheperBridge := BR_CONGESTED
else if bridgeClearing
    kheperBridge := BR_CLEARING

float handoffScore = 0.0
int handoffAbs = math.abs(kheperHandoff)
if handoffAbs == 1
    handoffScore := 54.0
else if handoffAbs == 2
    handoffScore := 86.0
else if handoffAbs == 3
    handoffScore := 82.0

float bridgeScore = 24.0
if kheperBridge == BR_BLOCKED
    bridgeScore := 86.0
else if kheperBridge == BR_CONGESTED
    bridgeScore := 68.0
else if kheperBridge == BR_CLEARING
    bridgeScore := 62.0

float velocityScore = f_clamp100(velocityAbs * 140.0)
if kheperVelocity == 2
    velocityScore := math.max(velocityScore, 72.0)
else if kheperVelocity == -2
    velocityScore := math.max(velocityScore, 54.0)
else if kheperVelocity == -1
    velocityScore := math.max(velocityScore, 48.0)

float alignmentScore = 0.0
if moveDir != DIR_FLAT and htfTrendDir != DIR_FLAT
    alignmentScore := moveDir == htfTrendDir ? 38.0 : 64.0

float kheperScore = f_clamp100(math.max(math.max(gatePressureScore, handoffScore), math.max(bridgeScore, math.max(velocityScore, alignmentScore))))

//==================================================
// Cause TF and state
//==================================================
bool currentTfCause = math.abs(impulseAtr) >= impulseAtrMin * 1.8 and htfTrendDir == DIR_FLAT
bool nextHtfCause = (upperReject or lowerReject or bridgeConflict) and nearAnyGate
bool bridgeCause = bridgeCongested or bridgeBlocked
bool velocityCause = (kheperVelocity == 2 or kheperVelocity == -2) and not nearAnyGate

int causeCount = 0
if currentTfCause
    causeCount += 1
if nextHtfCause
    causeCount += 1
if bridgeCause
    causeCount += 1
if velocityCause
    causeCount += 1

int kheperCauseTf = CAUSE_NONE
if causeCount > 1
    kheperCauseTf := CAUSE_MIXED
else if nextHtfCause
    kheperCauseTf := CAUSE_NEXT_HTF
else if bridgeCause
    kheperCauseTf := CAUSE_BRIDGE
else if currentTfCause
    kheperCauseTf := CAUSE_CURRENT_TF
else if velocityCause
    kheperCauseTf := CAUSE_VELOCITY

int kheperState = ST_IDLE
if kheperBridge == BR_BLOCKED
    kheperState := ST_BLOCK
else if handoffAbs == 2
    kheperState := ST_HANDOFF
else if handoffAbs == 1 or nearAnyGate
    kheperState := ST_TEST
else if kheperScore >= activeMin or kheperBridge == BR_CONGESTED or kheperBridge == BR_CLEARING
    kheperState := ST_APPROACH

int kheperActiveFlag = (kheperState != ST_IDLE or kheperScore >= activeMin) ? 1 : 0

int kheperDir = DIR_FLAT
if kheperHandoff != 0
    kheperDir := kheperHandoff > 0 ? DIR_UP : DIR_DN
else if moveDir != DIR_FLAT
    kheperDir := moveDir
else
    kheperDir := htfTrendDir

string kheperPolicy = "OBSERVE"
if kheperBridge == BR_BLOCKED
    kheperPolicy := "GATE_BLOCKED"
else if kheperBridge == BR_CONGESTED
    kheperPolicy := "BRIDGE_CONGESTED"
else if handoffAbs == 2
    kheperPolicy := "HANDOFF_ACCEPTED"
else if kheperBridge == BR_CLEARING
    kheperPolicy := "BRIDGE_CLEARING"
else if handoffAbs == 1
    kheperPolicy := "HANDOFF_TEST"
else if kheperVelocity == 2
    kheperPolicy := "VELOCITY_ACCEL"
else if kheperVelocity == -2
    kheperPolicy := "VELOCITY_STALL"
else if kheperBridge == BR_OPEN
    kheperPolicy := "GATE_OPEN"

//==================================================
// Events
//==================================================
bool prevNearUpperGate = bar_index > 0 ? nearUpperGate[1] : false
bool prevNearLowerGate = bar_index > 0 ? nearLowerGate[1] : false
bool prevUpAccepted = bar_index > 0 ? upAccepted[1] : false
bool prevDnAccepted = bar_index > 0 ? dnAccepted[1] : false
bool prevUpperReject = bar_index > 0 ? upperReject[1] : false
bool prevLowerReject = bar_index > 0 ? lowerReject[1] : false
bool prevBridgeCongested = bar_index > 0 ? bridgeCongested[1] : false
bool prevBridgeClearing = bar_index > 0 ? bridgeClearing[1] : false
int prevVelocity = bar_index > 0 ? kheperVelocity[1] : 0
int prevActiveFlag = bar_index > 0 ? kheperActiveFlag[1] : 0

bool upperGateOn = nearUpperGate and not prevNearUpperGate
bool lowerGateOn = nearLowerGate and not prevNearLowerGate
bool handoffUpOn = upAccepted and not prevUpAccepted
bool handoffDnOn = dnAccepted and not prevDnAccepted
bool upperBlockOn = upperReject and not prevUpperReject
bool lowerBlockOn = lowerReject and not prevLowerReject
bool congestedOn = bridgeCongested and not prevBridgeCongested
bool clearingOn = bridgeClearing and not prevBridgeClearing
bool accelOn = kheperVelocity == 2 and prevVelocity != 2
bool stallOn = kheperVelocity == -2 and prevVelocity != -2
bool activeOn = kheperActiveFlag == 1 and prevActiveFlag == 0

int kheperEvent = EC_NONE
if handoffUpOn
    kheperEvent := EC_HANDOFF_UP
else if handoffDnOn
    kheperEvent := EC_HANDOFF_DN
else if upperBlockOn
    kheperEvent := EC_UPPER_BLOCK
else if lowerBlockOn
    kheperEvent := EC_LOWER_BLOCK
else if congestedOn
    kheperEvent := EC_CONGESTED
else if clearingOn
    kheperEvent := EC_CLEARING
else if upperGateOn
    kheperEvent := EC_UPPER_GATE
else if lowerGateOn
    kheperEvent := EC_LOWER_GATE
else if accelOn
    kheperEvent := EC_ACCEL
else if stallOn
    kheperEvent := EC_STALL
else if activeOn
    kheperEvent := EC_ACTIVE_ON

float eventSeverity = f_clamp100(math.max(kheperScore, handoffScore))

//==================================================
// GV3 alert bus
//==================================================
string kheperAlertJson = f_kheperAlertJson(kheperEvent, eventSeverity, kheperPolicy, htfPosition, kheperHandoff, kheperBridge, kheperVelocity, kheperCauseTf, nextTf)

if useJsonAlerts and kheperEvent != EC_NONE
    if alertFrequencyMode == "BAR_CLOSE"
        if barstate.isconfirmed
            alert(kheperAlertJson, alert.freq_once_per_bar_close)
    else
        alert(kheperAlertJson, alert.freq_once_per_bar)

//==================================================
// Compact GPT table
//==================================================
var table kheperTable = na
var string lastTableLayoutKey = ""

color colBg = color.rgb(12, 14, 18)
color colHead = color.rgb(230, 236, 242)
color colText = color.rgb(205, 212, 220)
color colSoft = color.rgb(145, 154, 166)
color colGood = color.rgb(70, 210, 145)
color colWarn = color.rgb(245, 185, 70)
color colBad = color.rgb(245, 95, 95)
color colInfo = color.rgb(90, 190, 240)
color colAccent = color.rgb(185, 150, 255)
color colFrame = color.new(color.white, 50)
color dirColor = kheperDir > 0 ? colGood : kheperDir < 0 ? colBad : colSoft
color stateColor = kheperState == ST_HANDOFF ? colGood : kheperState == ST_BLOCK ? colBad : kheperState == ST_TEST ? colWarn : kheperState == ST_APPROACH ? colInfo : colSoft
color eventColor = kheperEvent == EC_NONE ? colSoft : stateColor
