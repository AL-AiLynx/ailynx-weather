f_clamp(float x, float lo, float hi) =>
    math.min(hi, math.max(lo, x))

f_pack_abs(float pack) =>
    math.max(nz(pack, 0.0), 0.0)

f_dir(float pack) =>
    int(math.floor(f_pack_abs(pack) / 10000.0))

f_risk(float pack) =>
    int(math.floor(f_pack_abs(pack) / 1000.0)) % 10

f_event(float pack) =>
    int(math.floor(f_pack_abs(pack) / 10.0)) % 100

f_valid(float pack) =>
    int(math.round(f_pack_abs(pack))) % 10

f_linked(float value) =>
    not na(value) and math.abs(value - close) > 0.000001

f_score(float value, bool ok) =>
    ok ? f_clamp(nz(value, 0.0), 0.0, 100.0) : 0.0

f_pos(string p) =>
    out = position.top_right
    if p == "top_left"
        out := position.top_left
    else if p == "top_center"
        out := position.top_center
    else if p == "top_right"
        out := position.top_right
    else if p == "middle_left"
        out := position.middle_left
    else if p == "middle_center"
        out := position.middle_center
    else if p == "middle_right"
        out := position.middle_right
    else if p == "bottom_left"
        out := position.bottom_left
    else if p == "bottom_center"
        out := position.bottom_center
    else if p == "bottom_right"
        out := position.bottom_right
    out

    string out = "OFF"

        if not linked
            out := "NA"
        else if not valid
            out := "INVALID"

            out := f_dirText(dirCode) + " R" + str.tostring(riskCode) + " " + eventText + " S" + str.tostring(math.round(scoreValue)) + " V1"
