f_clamp(float x, float lo, float hi) =>
    math.min(hi, math.max(lo, x))

f_fmt(float price) =>
    str.tostring(price, format.mintick)

f_style(string opt) =>
    out = line.style_solid
    if opt == "DASHED"
        out := line.style_dashed
    else if opt == "DOTTED"
        out := line.style_dotted
    out

f_pos(string p) =>
    out = position.bottom_right
    if p == "top_left"
        out := position.top_left
    else if p == "top_center"
        out := position.top_center
    else if p == "top_right"
        out := position.top_right
    else if p == "middle_left"
        out := position.middle_left
    else if p == "middle_center"
        out := position.middle_center
    else if p == "middle_right"
        out := position.middle_right
    else if p == "bottom_left"
        out := position.bottom_left
    else if p == "bottom_center"
        out := position.bottom_center
    out

f_size(string s) =>
    out = size.small
    if s == "tiny"
        out := size.tiny
    else if s == "normal"
        out := size.normal
    else if s == "large"
        out := size.large
    out

f_slotName(int idx) =>
    string out = "U1"
    if idx == IDX_U2
        out := "U2"
    else if idx == IDX_D1
        out := "D1"
    else if idx == IDX_D2
        out := "D2"
    out

f_stateText(int stateCode) =>
    string out = "NONE"
    if stateCode == STATE_MEMORY
        out := "MEMORY"
    else if stateCode == STATE_TOUCH
        out := "TOUCH"
    else if stateCode == STATE_ABSORB
        out := "ABSORB"
    else if stateCode == STATE_REJECT
        out := "REJECT"
    else if stateCode == STATE_CLOSE_BREAK
        out := "CLOSE_BREAK"
    else if stateCode == STATE_RETEST_HOLD
        out := "RETEST_HOLD"
    else if stateCode == STATE_SHIFT
        out := "SHIFT"
    else if stateCode == STATE_NOISE
        out := "NOISE"
    out

f_eventText(int eventCode) =>
    string out = "NONE"
    if eventCode == 1
        out := "TOUCH"
    else if eventCode == 2
        out := "ABSORB_REJECT"
    else if eventCode == 3
        out := "CLOSE_BREAK"
    else if eventCode == 4
        out := "RETEST_HOLD"
    else if eventCode == 5
        out := "SHIFT"
    else if eventCode == 9
        out := "RESET"
    out

f_stateToMaat2Event(int stateCode) =>
    int out = MAAT2_WAS_EVENT_NONE
    if stateCode == STATE_MEMORY
        out := MAAT2_WAS_EVENT_MEMORY
    else if stateCode == STATE_TOUCH
        out := MAAT2_WAS_EVENT_TOUCH
    else if stateCode == STATE_ABSORB
        out := MAAT2_WAS_EVENT_ABSORB
    else if stateCode == STATE_REJECT
        out := MAAT2_WAS_EVENT_REJECT
    else if stateCode == STATE_CLOSE_BREAK
        out := MAAT2_WAS_EVENT_CLOSE_BREAK
    else if stateCode == STATE_RETEST_HOLD
        out := MAAT2_WAS_EVENT_RETEST_HOLD
    else if stateCode == STATE_SHIFT
        out := MAAT2_WAS_EVENT_SHIFT
    else if stateCode == STATE_NOISE
        out := MAAT2_WAS_EVENT_NOISE
    out

f_legacyEventToMaat2(int legacyEvent, int nearestState) =>
    int out = f_stateToMaat2Event(nearestState)
    if legacyEvent == 1
        out := MAAT2_WAS_EVENT_TOUCH
    else if legacyEvent == 2
        out := nearestState == STATE_REJECT ? MAAT2_WAS_EVENT_REJECT : MAAT2_WAS_EVENT_ABSORB
    else if legacyEvent == 3
        out := MAAT2_WAS_EVENT_CLOSE_BREAK
    else if legacyEvent == 4
        out := MAAT2_WAS_EVENT_RETEST_HOLD
    else if legacyEvent == 5
        out := MAAT2_WAS_EVENT_SHIFT
    else if legacyEvent == 9
        out := MAAT2_WAS_EVENT_NONE
    out

f_max_int(int a, int b) =>
    a > b ? a : b

f_confRisk(float conf) =>
    int out = 0
    if conf < retireConf + 5.0
        out := 3
    else if conf < 45.0
        out := 2
    else if conf < 60.0
        out := 1
    out

f_distRisk(float distPct) =>
    int out = 0
    if not na(distPct)
        if distPct <= mergePct
            out := 4
        else if distPct <= mergePct * 2.0
            out := 3
        else if distPct <= 1.0
            out := 2
        else if distPct <= 2.0
            out := 1
    out

f_stateRisk(int stateCode) =>
    int out = 0
    if stateCode == STATE_MEMORY
        out := 1
    else if stateCode == STATE_TOUCH
        out := 3
    else if stateCode == STATE_ABSORB or stateCode == STATE_REJECT
        out := 2
    else if stateCode == STATE_CLOSE_BREAK
        out := 3
    else if stateCode == STATE_RETEST_HOLD
        out := 1
    else if stateCode == STATE_SHIFT
        out := 4
    else if stateCode == STATE_NOISE
        out := 5
    out

f_eventRisk(int eventCode) =>
    int out = 0
    if eventCode == MAAT2_WAS_EVENT_MEMORY
        out := 1
    else if eventCode == MAAT2_WAS_EVENT_TOUCH
        out := 3
    else if eventCode == MAAT2_WAS_EVENT_ABSORB or eventCode == MAAT2_WAS_EVENT_REJECT
        out := 2
    else if eventCode == MAAT2_WAS_EVENT_CLOSE_BREAK
        out := 3
    else if eventCode == MAAT2_WAS_EVENT_RETEST_HOLD
        out := 1
    else if eventCode == MAAT2_WAS_EVENT_SHIFT
        out := 4
    else if eventCode == MAAT2_WAS_EVENT_NOISE
        out := 5
    out

f_delete_objects(int idx) =>
    line l = array.get(memLine, idx)
    label lb = array.get(memLabel, idx)
    if showVisuals and not na(l)
        line.delete(l)
    if showVisuals and not na(lb)
        label.delete(lb)
    array.set(memLine, idx, na)
    array.set(memLabel, idx, na)

f_empty_values(int idx) =>
    array.set(memPrice, idx, na)
    array.set(memState, idx, STATE_NONE)
    array.set(memConf, idx, 0.0)
    array.set(memTouches, idx, 0)
    array.set(memAge, idx, 0)
    array.set(memActive, idx, false)

f_clear_slot(int idx) =>
    f_delete_objects(idx)
    f_empty_values(idx)

f_add_conf(int idx, float delta) =>
    float nextConf = f_clamp(array.get(memConf, idx) + delta, 0.0, 100.0)
    array.set(memConf, idx, nextConf)
    nextConf

f_touch_slot(int idx, bool hasConfirmation) =>
    int nextTouches = array.get(memTouches, idx) + 1
    array.set(memTouches, idx, nextTouches)
    if not hasConfirmation
        f_add_conf(idx, -5.0)
    if nextTouches > 2 and nextTouches <= 5
        f_add_conf(idx, -10.0)
    nextTouches

f_move_slot(int fromIdx, int toIdx) =>
    if array.get(memActive, fromIdx)
        f_delete_objects(toIdx)
        array.set(memPrice, toIdx, array.get(memPrice, fromIdx))
        array.set(memState, toIdx, array.get(memState, fromIdx))
        array.set(memConf, toIdx, array.get(memConf, fromIdx))
        array.set(memTouches, toIdx, array.get(memTouches, fromIdx))
        array.set(memAge, toIdx, array.get(memAge, fromIdx))
        array.set(memActive, toIdx, true)
        array.set(memLine, toIdx, array.get(memLine, fromIdx))
        array.set(memLabel, toIdx, array.get(memLabel, fromIdx))
        array.set(memLine, fromIdx, na)
        array.set(memLabel, fromIdx, na)
        f_empty_values(fromIdx)
    else
        f_clear_slot(toIdx)

f_set_slot(int idx, float price, float conf) =>
    f_clear_slot(idx)
    array.set(memPrice, idx, price)
    array.set(memState, idx, STATE_MEMORY)
    array.set(memConf, idx, f_clamp(conf, 0.0, 100.0))
    array.set(memTouches, idx, 0)
    array.set(memAge, idx, 0)
    array.set(memActive, idx, true)

f_find_near(float price, int idxA, int idxB) =>
    int found = -1
    if not na(price)
        float pA = array.get(memPrice, idxA)
        if array.get(memActive, idxA) and not na(pA) and math.abs(pA) > 0.0
            float distA = math.abs(price - pA) / math.abs(pA) * 100.0
            if distA <= mergePct
                found := idxA
        if found < 0
            float pB = array.get(memPrice, idxB)
            if array.get(memActive, idxB) and not na(pB) and math.abs(pB) > 0.0
                float distB = math.abs(price - pB) / math.abs(pB) * 100.0
                if distB <= mergePct
                    found := idxB
    found

f_merge_slot(int idx, float price, float conf) =>
    float oldPrice = array.get(memPrice, idx)
    float mergedPrice = na(oldPrice) ? price : (oldPrice + price) * 0.5
    int oldState = array.get(memState, idx)
    array.set(memPrice, idx, mergedPrice)
    array.set(memAge, idx, 0)
    if oldState == STATE_NONE or oldState == STATE_NOISE
        array.set(memState, idx, STATE_MEMORY)
    array.set(memActive, idx, true)

f_apply_candidate(bool isUpper, float price, float conf) =>
    int primary = isUpper ? IDX_U1 : IDX_D1
    int previous = isUpper ? IDX_U2 : IDX_D2
    int nearIdx = f_find_near(price, primary, previous)
    if nearIdx >= 0
        f_merge_slot(nearIdx, price, conf)
    else
        f_move_slot(primary, previous)
        f_set_slot(primary, price, conf)

f_refresh_slot(int idx) =>
    bool isActive = array.get(memActive, idx)
    float price = array.get(memPrice, idx)
    line l = array.get(memLine, idx)
    label lb = array.get(memLabel, idx)
    if not isActive or na(price)
        f_delete_objects(idx)
    else
        bool isUpper = idx == IDX_U1 or idx == IDX_U2
        bool isPrimary = idx == IDX_U1 or idx == IDX_D1
        color baseColor = isUpper ? upperColor : lowerColor
        int lineTransp = isPrimary ? activeTransp : previousTransp
        int lineWidth = isPrimary ? activeLineWidth : previousLineWidth
        string styleOpt = isPrimary ? activeLineStyle : previousLineStyle
        string labelText = f_slotName(idx) + " " + f_stateText(array.get(memState, idx)) + " S" + str.tostring(int(math.round(array.get(memConf, idx)))) + " " + f_fmt(price)
        if showLines
            if na(l)
                l := line.new(bar_index, price, bar_index + 1, price, xloc=xloc.bar_index, extend=extend.right, color=color.new(baseColor, lineTransp), width=lineWidth, style=f_style(styleOpt))
            else
                line.set_y1(l, price)
                line.set_y2(l, price)
                line.set_color(l, color.new(baseColor, lineTransp))
                line.set_width(l, lineWidth)
                line.set_style(l, f_style(styleOpt))
            array.set(memLine, idx, l)
            if showLabels
                if na(lb)
                    lb := label.new(bar_index + labelOffsetBars, price, labelText, xloc=xloc.bar_index, yloc=yloc.price, style=label.style_label_left, textcolor=color.white, color=color.new(baseColor, isPrimary ? 0 : previousTransp), size=size.small)
                else
                    label.set_x(lb, bar_index + labelOffsetBars)
                    label.set_y(lb, price)
                    label.set_text(lb, labelText)
                    label.set_color(lb, color.new(baseColor, isPrimary ? 0 : previousTransp))
                    label.set_textcolor(lb, color.white)
                array.set(memLabel, idx, lb)
            else
                if not na(lb)
                    label.delete(lb)
                    lb := na
                    array.set(memLabel, idx, na)
        else
            f_delete_objects(idx)

f_degrade_slot(int idx) =>
    bool retired = false
    if array.get(memActive, idx)
        int nextAge = array.get(memAge, idx) + 1
        array.set(memAge, idx, nextAge)
        if nextAge % 48 == 0 and nextAge <= 192
            f_add_conf(idx, -5.0)
        if array.get(memConf, idx) < retireConf
            f_clear_slot(idx)
            retired := true
    retired

f_ha_pack() =>
    float ph = ta.pivothigh(high, pivotLeft, pivotRight)[1]
    float pl = ta.pivotlow(low, pivotLeft, pivotRight)[1]
    bool phOk = not na(ph)
    bool plOk = not na(pl)
    float hiRef = ta.highest(high[2], dispLen)
    float loRef = ta.lowest(low[2], dispLen)
    bool dispHi = not na(hiRef) and high[1] > hiRef
    bool dispLo = not na(loRef) and low[1] < loRef
    float upPrice = phOk ? ph : high[1]
    float dnPrice = plOk ? pl : low[1]
    float upWick = phOk ? high[pivotRight + 1] - math.max(open[pivotRight + 1], close[pivotRight + 1]) : high[1] - math.max(open[1], close[1])
    float dnWick = plOk ? math.min(open[pivotRight + 1], close[pivotRight + 1]) - low[pivotRight + 1] : math.min(open[1], close[1]) - low[1]
    float upBody = phOk ? math.abs(close[pivotRight + 1] - open[pivotRight + 1]) : math.abs(close[1] - open[1])
    float dnBody = plOk ? math.abs(close[pivotRight + 1] - open[pivotRight + 1]) : math.abs(close[1] - open[1])
    float upAtr = phOk ? ta.atr(atrLen)[pivotRight + 1] : ta.atr(atrLen)[1]
    float dnAtr = plOk ? ta.atr(atrLen)[pivotRight + 1] : ta.atr(atrLen)[1]
    [open[1], high[1], low[1], close[1], time[1], ta.atr(atrLen)[1], upPrice, upWick, upBody, upAtr, phOk, dispHi, dnPrice, dnWick, dnBody, dnAtr, plOk, dispLo]

f_process_line(int idx, bool isUpper) =>
    int eventCode = 0
    if array.get(memActive, idx)
        float p = array.get(memPrice, idx)
        int priorState = array.get(memState, idx)
        bool wasBroken = priorState == STATE_CLOSE_BREAK or priorState == STATE_SHIFT
        bool lineTouched = isUpper ? math.max(haHigh4h, normalHigh4h) >= p : math.min(haLow4h, normalLow4h) <= p
        bool closeBreak = isUpper ? confirmClose4h > p : confirmClose4h < p
        bool closeReturn = isUpper ? lineTouched and confirmClose4h < p : lineTouched and confirmClose4h > p
        bool retestHold = wasBroken and (isUpper ? math.min(haLow4h, normalLow4h) <= p and confirmClose4h >= p : math.max(haHigh4h, normalHigh4h) >= p and confirmClose4h <= p)
        if retestHold
            f_touch_slot(idx, true)
            f_add_conf(idx, 10.0)
            array.set(memState, idx, STATE_RETEST_HOLD)
            eventCode := 4
        else if closeBreak and not wasBroken
            f_touch_slot(idx, true)
            bool strongShift = array.get(memConf, idx) >= strongScore
            bool flipShift = takeoverMode == "FLIP" and array.get(memConf, idx) >= strongScore - 10.0
            array.set(memState, idx, strongShift or flipShift ? STATE_SHIFT : STATE_CLOSE_BREAK)
            eventCode := strongShift or flipShift ? 5 : 3
        else if closeReturn
            f_touch_slot(idx, true)
            f_add_conf(idx, 10.0)
            bool rejectStyle = isUpper ? haClose4h < haOpen4h : haClose4h <= haOpen4h
            int reactionState = isUpper ? (rejectStyle ? STATE_REJECT : STATE_ABSORB) : (rejectStyle ? STATE_REJECT : STATE_ABSORB)
            array.set(memState, idx, reactionState)
            eventCode := 2
        else if lineTouched
            int touchN = f_touch_slot(idx, false)
            array.set(memState, idx, touchN > 3 ? STATE_NOISE : STATE_TOUCH)
            eventCode := 1
    eventCode
