f_clamp(float value, float low, float high) =>
    math.max(low, math.min(high, value))

f_hasBit(int mask, int bitValue) =>
    int(math.floor(float(mask) / float(bitValue))) % 2 == 1

f_autoParent(int minutes) =>
    minutes < 60 ? 60 : minutes < 120 ? 120 : minutes < 180 ? 180 : minutes < 240 ? 240 : minutes < 360 ? 360 : minutes < 480 ? 480 : minutes < 720 ? 720 : minutes < 1440 ? 1440 : minutes < 2880 ? 2880 : minutes < 5760 ? 5760 : 10080

f_position(string value) =>
    value == "top_left" ? position.top_left : value == "bottom_right" ? position.bottom_right : value == "bottom_left" ? position.bottom_left : position.top_right

f_size(string value) =>
    value == "tiny" ? size.tiny : value == "normal" ? size.normal : size.small

"FORCE_EXPRESSION" : code == TIME_STATE_FALSE_RELEASE ? "FALSE_RELEASE" : code == TIME_STATE_NOISE_ONLY ? "NOISE_ONLY" : "RESET_REQUIRED"

"WINDOW_OPEN" : code == TIME_RESULT_FORCE_EXPRESSION_CANDIDATE ? "FORCE_EXPRESSION_CANDIDATE" : code == TIME_RESULT_NOISE_ONLY ? "NOISE_ONLY" : code == TIME_RESULT_FALSE_RELEASE ?     string escaped = str.replace_all(value, "\\", "\\\\")
    escaped := str.replace_all(escaped, "\"", "\\\"")
    "\"" + escaped + "\""

    na(value) ? "null" : str.tostring(value)

    na(value) ? "null" : str.tostring(value)

    value ? "true" : "false"

int repackedState = parsedMagic * BRIDGE_MAGIC_MULT + parsedPhase * PHASE_MULT + parsedHubResult * RESULT_MULT + parsedRiskBand * RISK_BAND_MULT + parsedGuardMask * GUARD_MASK_MULT + parsedAlertMask * ALERT_MASK_MULT + parsedCoreState * CORE_STATE_MULT + parsedPeripheralState * PERIPH_STATE_MULT + parsedBridgeValid
bool stateRoundTripOk = stateInputAvailable and stateRangeOk and repackedState == stateRaw

int repackedMetrics = parsedStructure * METRIC_STRUCTURE_MULT + parsedForce * METRIC_FORCE_MULT + parsedWindow * METRIC_WINDOW_MULT + parsedRisk
bool metricsRoundTripOk = metricsInputAvailable and metricsRangeOk and repackedMetrics == metricsRaw

    statePart := 15.0
else if (parsedPhase == PHASE_WINDOW_OPEN and guardWindow) or ((parsedPhase == PHASE_FORCE_EVENT or parsedPhase == PHASE_ABSORB_CANDIDATE) and guardForce)
    statePart := 10.0
else if parsedPhase == PHASE_VOL_BUILD or parsedPhase == PHASE_VOL_TRIGGER_WAIT
    statePart := 6.0
else if engineBridgeValid and parsedHubResult == HUB_RESULT_WATCH
    statePart := 3.0

float noisePenalty = (guardNoise ? 12.0 : 0.0) + (alertNoise ? 8.0 : 0.0) + (guardReset ? 15.0 : 0.0) + (alertReset ? 10.0 : 0.0)
noisePenalty := math.min(30.0, noisePenalty)
float rawTimeScore = f_clamp(structurePart + forcePart + windowPart + statePart - riskPenalty - noisePenalty, 0.0, 100.0)
float cappedTimeScore = engineBridgeValid ? rawTimeScore : 0.0
if systemInvalid
    cappedTimeScore := math.min(cappedTimeScore, 35.0)
if systemPartial
    cappedTimeScore := math.min(cappedTimeScore, 55.0)
if parsedForce < 20
    cappedTimeScore := math.min(cappedTimeScore, 50.0)
if parsedRisk >= 75
    cappedTimeScore := math.min(cappedTimeScore, 40.0)
if guardReset
    cappedTimeScore := math.min(cappedTimeScore, 35.0)
int timeScore = int(math.round(f_clamp(cappedTimeScore, 0.0, 100.0)))

float rawNoiseScore = float(parsedRisk) * 0.65 + (guardNoise ? 15.0 : 0.0) + (alertNoise ? 10.0 : 0.0) + (guardReset ? 15.0 : 0.0) + (systemPartial ? 10.0 : 0.0) + (systemInvalid ? 20.0 : 0.0)
