f_sizeFromScale(int scale) =>
    result = size.small
    if scale <= 2
        result := size.tiny
    else if scale <= 5
        result := size.small
    else if scale <= 7
        result := size.normal
    else if scale <= 9
        result := size.large
    else
        result := size.huge
    result

f_riskCode(float chopPenalty, float squeezeBoost, float hetMove, float ctxUp, float ctxDown) =>
    int result = 0
    if ctxUp >= 35 and ctxDown >= 35 and math.abs(ctxUp - ctxDown) <= 10
        result := 3
    if chopPenalty > 0.7
        result := math.max(result, 3)
    else if chopPenalty > 0.4
        result := math.max(result, 2)
    if squeezeBoost > 0.65
        result := math.max(result, 4)
    if math.abs(hetMove) > 0.75
        result := math.max(result, 4)
    result

f_band(float v) =>
    float x = f_clamp(v, 0.0, 100.0)
    string out = "NONE"
    if x >= 80.0
        out := "EXTREME"
    else if x >= 60.0
        out := "HIGH"
    else if x >= 40.0
        out := "MID"
    else if x >= 20.0
        out := "LOW"
    out

    string out = "FLAT_NONE"

    else if v < 0.0
        out := "DN_" + f_band(math.abs(v) * 100.0)
    out

f_ob(int pivLen) =>
    var float bullTopMem = na
    var float bullBotMem = na
    var float bearTopMem = na
    var float bearBotMem = na
    float pl = ta.pivotlow(low, pivLen, pivLen)
    float ph = ta.pivothigh(high, pivLen, pivLen)
    if not na(pl)
        bullTopMem := math.max(open[pivLen], close[pivLen])
        bullBotMem := low[pivLen]
    if not na(ph)
        bearTopMem := high[pivLen]
        bearBotMem := math.min(open[pivLen], close[pivLen])
    [bullTopMem, bullBotMem, bearTopMem, bearBotMem]

f_wenutStats(int days, string zone) =>
    int currentHour = hour(time, zone)
    int bullCnt = 0
    int bearCnt = 0
    int sampleCnt = 0
    int barsBack = math.max(days * 24, 24)
    for i = 1 to barsBack
        if not na(close[i]) and not na(time[i]) and hour(time[i], zone) == currentHour
            sampleCnt += 1
            bullCnt += close[i] > open[i] ? 1 : 0
            bearCnt += close[i] < open[i] ? 1 : 0
    float bullPct = sampleCnt > 0 ? float(bullCnt) / float(sampleCnt) * 100.0 : 0.0
    float bearPct = sampleCnt > 0 ? float(bearCnt) / float(sampleCnt) * 100.0 : 0.0
    [bullPct, bearPct, float(currentHour)]
