f_tfLabel() =>
    string out = timeframe.period
    if timeframe.isseconds
        out := str.tostring(timeframe.multiplier) + "S"
    else if timeframe.isminutes
        int m = timeframe.multiplier
        if m % 60 == 0
            out := str.tostring(int(m / 60)) + "H"
        else
            out := str.tostring(m) + "M"
    else if timeframe.isdaily
        out := timeframe.multiplier == 1 ? "1D" : str.tostring(timeframe.multiplier) + "D"
    else if timeframe.isweekly
        out := timeframe.multiplier == 1 ? "1W" : str.tostring(timeframe.multiplier) + "W"
    else if timeframe.ismonthly
        out := timeframe.multiplier == 1 ? "1MO" : str.tostring(timeframe.multiplier) + "MO"
    out

f_pad2(int n) =>
    n < 10 ? "0" + str.tostring(n) : str.tostring(n)

// v1.2: 잔여초를 전역에서 계산해 색상 판정에 재사용 (문자열 비교 버그 수정)
float cdRawSec = na(time_close) ? 0.0 : math.max(0.0, (time_close - timenow) / 1000.0)

f_countdownText(float rawSec) =>
    int totalSec = int(math.floor(rawSec))
    int h = int(math.floor(totalSec / 3600))
    int m = int(math.floor((totalSec % 3600) / 60))
    int s = totalSec % 60
    string out = ""
    if h > 0
        out := f_pad2(h) + ":" + f_pad2(m) + ":" + f_pad2(s)
    else
        out := f_pad2(m) + ":" + f_pad2(s)
    out

f_standardSymbol() =>
    string px = syminfo.prefix
    string tk = syminfo.ticker
    string out = px != "" ? px + ":" + tk : tk
    out

f_symbolText() =>
    string standard = f_standardSymbol()
    string out = standard
    if symbolMode == "내부ID"
        out := syminfo.tickerid
    else if symbolMode == "커스텀"
        out := customSymbolText != "" ? customSymbolText : standard
    else
        out := standard
    out := customPrefix != "" ? customPrefix + " " + out : out
    out

// 헬퍼 | Formatting
//==================================================
f_pct(float v) =>
    na(v) ? "na" : str.tostring(v, "#.##") + "%"

f_priceK(float v) =>
    na(v) ? "na" : str.tostring(v / 1000.0, "#.##") + "K"

f_rangeK(float lo, float hi) =>
    f_priceK(lo) + "~" + f_priceK(hi)

f_durText(float days) =>
    string out = ""
    if na(days)
        out := ""
    else if days < 1
        out := "<1D"
    else if days < 100
        out := str.tostring(days, "#") + "D"
    else
        out := str.tostring(days / 30.44, "#.#") + "MO"
    out

// v1.2: D 구간 이중 표기 (↑재탈환선 / ↓다음 방어선)
f_levelText(string gateCode) =>
    string out = ""
    if gateCode == "G1_TEST" or gateCode == "G1_WAIT" or gateCode == "G1_RECLAIM"
        out := f_rangeK(g1Low, g1High)
    else if gateCode == "G2_TEST" or gateCode == "G2_RECLAIM"
        out := f_rangeK(g2Low, g2High)
    else if gateCode == "G3_TEST" or gateCode == "G3_RECLAIM"
        out := f_rangeK(g3Low, g3High)
    else if gateCode == "G4_ABOVE"
        out := f_priceK(g4Level)
    else if gateCode == "D1_WATCH"
        out := "↑" + f_priceK(d1Level) + " ↓" + f_priceK(d2Level)
    else if gateCode == "D2_WATCH"
        out := "↑" + f_priceK(d2Level) + " ↓" + f_priceK(d3Level)
    else if gateCode == "D3_WATCH"
        out := "↑" + f_priceK(d3Level) + " ↓" + f_priceK(d4High)
    else if gateCode == "D4_LIFELINE"
        out := f_rangeK(d4Low, d4High) + " ↓" + f_priceK(d4SweepHigh)
    else if gateCode == "D4_BROKEN"
        out := "↑" + f_priceK(d4Low) + " ↓" + f_priceK(d4SweepHigh)
    else if gateCode == "D4_SWEEP"
        out := f_rangeK(d4SweepLow, d4SweepHigh)
    else if gateCode == "BELOW_SWEEP"
        out := "↓" + f_priceK(d4SweepLow)
    out

f_m2Code(float pct) =>
    string out = "UNK"
    if na(pct)
        out := "UNK"
    else if pct >= m2ExpTh
        out := "EXP"
    else if pct <= m2ConTh
        out := "CON"
    else if pct > 0
        out := "SLOW"
    else
        out := "NEU"
    out

f_btcM2Code(string m2c, float bPct) =>
    string out = "UNK"
    if m2c == "UNK" or na(bPct)
        out := "UNK"
    else if m2c == "EXP" and bPct <= btcDownTh
        out := "DECPL"
    else if m2c == "EXP" and bPct >= btcUpTh
        out := "RECPL"
    else if (m2c == "EXP" or m2c == "SLOW") and bPct > 0
        out := "CPL"
    else if m2c == "CON" and bPct < 0
        out := "RISK"
    else
        out := "NEU"
    out

f_gateCode(float p) =>
    string out = "UNK"
    if na(p)
        out := "UNK"
    else if p >= g4Level
        out := "G4_ABOVE"
    else if p >= g3High
        out := "G3_RECLAIM"
    else if p >= g3Low
        out := "G3_TEST"
    else if p >= g2High
        out := "G2_RECLAIM"
    else if p >= g2Low
        out := "G2_TEST"
    else if p >= g1High
        out := "G1_RECLAIM"
    else if p >= g1Low
        out := "G1_TEST"
    else if p >= d1Level
        out := "G1_WAIT"
    else if p >= d2Level
        out := "D1_WATCH"
    else if p >= d3Level
        out := "D2_WATCH"
    else if p > d4High
        out := "D3_WATCH"
    else if p >= d4Low and p <= d4High
        out := "D4_LIFELINE"
    else if p > d4SweepHigh and p < d4Low
        out := "D4_BROKEN"
    else if p >= d4SweepLow and p <= d4SweepHigh
        out := "D4_SWEEP"
    else
        out := "BELOW_SWEEP"
    out

f_leanCode(string m2c, string bmc, string gc) =>
    string out = "NEUTRAL"
    bool gateRecovered = gc == "G2_RECLAIM" or gc == "G3_TEST" or gc == "G3_RECLAIM" or gc == "G4_ABOVE"
    bool dangerGate = gc == "D1_WATCH" or gc == "D2_WATCH" or gc == "D3_WATCH" or gc == "D4_LIFELINE" or gc == "D4_BROKEN" or gc == "D4_SWEEP" or gc == "BELOW_SWEEP"
    if dangerGate
        out := "RISK_OFF"
    else if m2c == "EXP" and (bmc == "RECPL" or bmc == "CPL") and gateRecovered
        out := "FUEL_ON_IGNITION_ON"
    else if m2c == "EXP" and bmc == "RECPL"
        out := "FUEL_ON_IGNITION_TRY"
    else if m2c == "EXP" and bmc == "DECPL"
        out := "FUEL_ON_IGNITION_OFF"
    else if m2c == "CON"
        out := "FUEL_LOW"
    else
        out := "NEUTRAL"
    out
