// ------------------------------------------------------------------------
// 이 지표는 실제 호가창/체결 테이프 기반 Bid/Ask Delta가 아닙니다.
// TradingView Pine에서 접근 가능한 OHLCV와 lower timeframe 배열을 이용해
// 매수·매도 압력을 추정하는 "LTF / PROXY 기반 거래량 분해 센서"입니다.
//
// v2 개선 핵심
//  1) 이름/설명 정리: TRUE 체결량이 아니라 TRUE FLOW PROXY로 명시
//  2) 분해 모드 추가: 차트 LTF 비율 적용 / 거래소별 PROXY 가중 분해 선택
//  3) 거래량 기준 추가: BASE / QUOTE / NORMALIZED
//  4) 숨은물량 boolean + deceptionScore 0~100 점수화
//  5) 흡수·클라이맥스 세분화: START / EXHAUST / TRAP / ABSORB
//  6) Spot / Perp / Binance Dominance 출력
//  7) alert() JSON 추가 + 기존 alertcondition 유지
//
// MAAT BUS 기본 출력
//  DIR_CODE   : -1 매도 흐름 / 0 중립 / 1 매수 흐름
//  SCORE      : 0~100 매수 우위도, 50 균형
//  RISK_CODE  : 0 안정 / 1 관찰 / 2 숨은물량 / 3 흡수 / 4 트랩 / 5 클라이맥스
//  EVENT_CODE : 0 없음 / 1 방향 / 5 클라이맥스 / 6 흡수 / 7 숨은물량 / 9 강한 거래량
//
// MAAT2 Compact BUS 추가 출력
//  PACK_CODE   : dirPack * 10000 + risk * 1000 + event * 10 + valid
//  TV_SCORE    : 방향 지시가 아니라 힘 발현 후보 강도
//  DETAIL_CODE : Hub/GPT 브리핑용 세부 이벤트 압축
// ════════════════════════════════════════════════════════════════════════

f_clamp(float x, float lo, float hi) =>
    math.max(lo, math.min(hi, x))

f_safe(float x) =>
    na(x) ? 0.0 : x

f_ok(float x) =>
    not na(x) and x > 0

f_ratio_proxy(float o, float h, float l, float c) =>
    float bp = (c - l) + (h - o)
    float sp = (h - c) + (o - l)
    float tp = bp + sp
    f_clamp(tp > 0 ? bp / tp : 0.5, 0.0, 1.0)

f_fmt_vol(float v) =>
    float av = math.abs(v)
    string sign = v < 0 ? "-" : ""
    string t = str.tostring(v, "#.##")
    if av >= 1e12
        t := sign + str.tostring(av / 1e12, "#.##") + "T"
    else if av >= 1e9
        t := sign + str.tostring(av / 1e9, "#.##") + "B"
    else if av >= 1e6
        t := sign + str.tostring(av / 1e6, "#.##") + "M"
    else if av >= 1e3
        t := sign + str.tostring(av / 1e3, "#.##") + "K"
    t

f_tsize(int n) =>
    n <= 2 ? size.tiny : n <= 6 ? size.small : n <= 8 ? size.normal : size.large

f_pos(string o) =>
    switch o
        "좌상" => position.top_left
        "상중" => position.top_center
        "우상" => position.top_right
        "좌중" => position.middle_left
        "정중" => position.middle_center
        "우중" => position.middle_right
        "좌하" => position.bottom_left
        "하중" => position.bottom_center
        => position.bottom_right

f_ltf_res(string mode) =>
    string r = "1"
    if mode == "자동"
        int s = timeframe.in_seconds(timeframe.period)
        if s <= 60
            r := "5S"
        else if s <= 300
            r := "30S"
        else if s <= 3600
            r := "1"
        else if s <= 14400
            r := "5"
        else if s <= 86400
            r := "15"
        else
            r := "60"
    else if mode == "1초"
        r := "1S"
    else if mode == "5초"
        r := "5S"
    else if mode == "10초"
        r := "10S"
    else if mode == "30초"
        r := "30S"
    else if mode == "1분"
        r := "1"
    else if mode == "3분"
        r := "3"
    else if mode == "5분"
        r := "5"
    else if mode == "15분"
        r := "15"
    r

f_sec(string sym) =>
    request.security(sym, timeframe.period, [open, high, low, close, volume], ignore_invalid_symbol=true)

int nIntr = array.size(ltfVolm)
if nIntr > 0
    for i = 0 to nIntr - 1
        float cc = array.get(ltfClose, i)
        float oo = array.get(ltfOpen, i)
        float vv = array.get(ltfVolm, i)
        if not (na(cc) or na(oo) or na(vv))
            if cc >= oo
                buyVolLtf += vv
            else
                sellVolLtf += vv

float ltfTot = buyVolLtf + sellVolLtf
int chartSec = timeframe.in_seconds(timeframe.period)
int ltfSec = timeframe.in_seconds(ltfRes)
bool ltfSmaller = ltfSec > 0 and chartSec > 0 and ltfSec < chartSec
bool ltfOk = useLTF and ltfSmaller and nIntr > 0 and ltfTot > 0
float buyRatioLtf = ltfOk ? buyVolLtf / ltfTot : na

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// Final split
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
float buyEx = w1*r1 + w2*r2 + w3*r3 + w4*r4 + w5*r5 + w6*r6 + w7*r7 + w8*r8 + w9*r9 + wC1*rC1 + wC2*rC2 + wC3*rC3
float sellEx = aggVol - buyEx
float buyRatioEx = aggVol > 0 ? buyEx / aggVol : na

float chartRatio = ltfOk ? buyRatioLtf : r0
bool useExchangeProxy = multiOk and splitMode == "거래소별 PROXY"
float buyRatio = useExchangeProxy ? buyRatioEx : chartRatio
buyRatio := f_clamp(nz(buyRatio, 0.5), 0.0, 1.0)
float sellRatio = 1.0 - buyRatio

float totalVol = totalVolRaw
float buyVol = totalVol * buyRatio
float sellVol = totalVol * sellRatio
float delta = buyVol - sellVol
float buyPct = buyRatio * 100.0
float sellPct = sellRatio * 100.0
float volAvg = ta.sma(totalVol, avgLen)
float relVol = volAvg > 0 ? totalVol / volAvg : 0.0

float dirBasis = usePrevColor ? close[1] : open
bool priceUp = close >= dirBasis
bool priceDn = close < dirBasis

string srcLabel = useExchangeProxy ? "EX-PROXY" : ltfOk ? "CHART-LTF " + ltfRes : "CHART-PROXY"

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 04. Deception / hidden flow score
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
deceptSoftTh = input.float(42.0, "숨은물량 시작 임계 %", minval=20.0, maxval=55.0, step=1.0, group=grpDecept, display=display.none)
deceptHardTh = input.float(60.0, "강한 숨은물량 임계 %", minval=45.0, maxval=90.0, step=1.0, group=grpDecept, display=display.none)
deceptMinRel = input.float(0.8, "최소 상대거래량", minval=0.0, step=0.1, group=grpDecept, display=display.none)
lieRelWeight = input.float(18.0, "상대거래량 점수 가중", minval=0.0, maxval=40.0, step=1.0, group=grpDecept, display=display.none)
lieWickWeight = input.float(12.0, "꼬리 점수 가중", minval=0.0, maxval=30.0, step=1.0, group=grpDecept, display=display.none)

float rangeSize = high - low
float bodySize = math.abs(close - open)
float bodyRatio = rangeSize > 0 ? bodySize / rangeSize : 0.0
float upperWick = high - math.max(open, close)
float lowerWick = math.min(open, close) - low
float upperWickRatio = rangeSize > 0 ? upperWick / rangeSize : 0.0
float lowerWickRatio = rangeSize > 0 ? lowerWick / rangeSize : 0.0

float oppositePct = priceUp ? sellPct : priceDn ? buyPct : 50.0
float lieBase = f_clamp((oppositePct - deceptSoftTh) / math.max(1.0, 100.0 - deceptSoftTh) * 100.0, 0.0, 100.0)
float lieRel = f_clamp((relVol - deceptMinRel) / 2.0 * lieRelWeight, 0.0, lieRelWeight)
float lieWick = priceUp ? upperWickRatio * lieWickWeight : priceDn ? lowerWickRatio * lieWickWeight : 0.0
float cvdConflictBoost = (priceUp and delta < 0) or (priceDn and delta > 0) ? 10.0 : 0.0
float deceptionScore = f_clamp(lieBase + lieRel + lieWick + cvdConflictBoost, 0.0, 100.0)

bool relEnough = relVol >= deceptMinRel
bool deceptSell = priceUp and sellPct >= deceptSoftTh and relEnough
bool deceptBuy = priceDn and buyPct >= deceptSoftTh and relEnough
bool hardDeceptSell = priceUp and sellPct >= deceptHardTh and relEnough
bool hardDeceptBuy = priceDn and buyPct >= deceptHardTh and relEnough
bool hiddenFlow = deceptSell or deceptBuy

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 05. Absorption / Climax detail
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
absorbVolMult = input.float(1.6, "흡수 거래량 기준", minval=1.0, step=0.1, group=grpAbsorb, display=display.none)
absorbBodyMax = input.float(0.38, "흡수 최대 몸통비율", minval=0.1, maxval=0.8, step=0.01, group=grpAbsorb, display=display.none)
absorbWickMin = input.float(0.22, "흡수 최소 꼬리비율", minval=0.0, maxval=0.8, step=0.01, group=grpAbsorb, display=display.none)
sweepLen = input.int(12, "스윕/돌파 확인 기간", minval=3, maxval=100, group=grpAbsorb, display=display.none)
climaxLen = input.int(60, "클라이맥스 기준 기간", minval=10, maxval=300, group=grpAbsorb, display=display.none)
climaxPct = input.float(0.90, "클라이맥스 거래량 기준", minval=0.5, maxval=1.0, step=0.05, group=grpAbsorb, display=display.none)
climaxDelta = input.float(62.0, "클라이맥스 방향 비중 %", minval=50.0, maxval=90.0, step=1.0, group=grpAbsorb, display=display.none)

bool isBigVol = relVol >= absorbVolMult
bool isSmallBody = bodyRatio <= absorbBodyMax
float prevLow = ta.lowest(low[1], sweepLen)
float prevHigh = ta.highest(high[1], sweepLen)
bool sweepLow = low < prevLow and close > prevLow
bool sweepHigh = high > prevHigh and close < prevHigh
bool breakoutUp = close > prevHigh
bool breakoutDown = close < prevLow

bool buyAbsorb = isBigVol and buyPct >= 50.0 and (isSmallBody or lowerWickRatio >= absorbWickMin) and (priceDn or sweepLow or lowerWickRatio > upperWickRatio)
bool sellAbsorb = isBigVol and sellPct >= 50.0 and (isSmallBody or upperWickRatio >= absorbWickMin) and (priceUp or sweepHigh or upperWickRatio > lowerWickRatio)

float totalVolHighest = ta.highest(totalVol, climaxLen)
bool climaxHi = totalVol >= totalVolHighest * climaxPct
bool buyDominant = buyPct >= climaxDelta
bool sellDominant = sellPct >= climaxDelta

bool buyClimaxStart = climaxHi and priceUp and buyDominant and breakoutUp and bodyRatio >= 0.45
bool sellClimaxStart = climaxHi and priceDn and sellDominant and breakoutDown and bodyRatio >= 0.45
bool buyClimaxExhaust = climaxHi and priceUp and buyDominant and upperWickRatio >= absorbWickMin and not breakoutUp
bool sellClimaxExhaust = climaxHi and priceDn and sellDominant and lowerWickRatio >= absorbWickMin and not breakoutDown
bool buyClimaxTrap = climaxHi and buyDominant and sweepHigh
bool sellClimaxTrap = climaxHi and sellDominant and sweepLow

bool buyClimax = buyClimaxStart or buyClimaxExhaust or buyClimaxTrap
bool sellClimax = sellClimaxStart or sellClimaxExhaust or sellClimaxTrap
bool anyClimax = buyClimax or sellClimax
bool anyAbsorb = buyAbsorb or sellAbsorb
bool anyTrap = buyClimaxTrap or sellClimaxTrap or hardDeceptBuy or hardDeceptSell

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 06. CVD
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
cvdLen = input.int(50, "CVD 누적 기간", minval=5, maxval=500, group=grpCVD, display=display.none)
cvdNormLen = input.int(200, "CVD 정규화 기간", minval=20, maxval=1000, group=grpCVD, display=display.none)
showCVD = input.bool(true, "CVD 라인 표시", group=grpCVD, display=display.none)
cvdScale = input.float(0.30, "CVD 표시 스케일", minval=0.05, maxval=2.0, step=0.05, group=grpCVD, display=display.none)

float cvdRoll = math.sum(delta, cvdLen)
float cvdMean = ta.sma(cvdRoll, cvdNormLen)
float cvdStd = ta.stdev(cvdRoll, cvdNormLen)
float cvdZ = cvdStd > 0 ? (cvdRoll - cvdMean) / cvdStd : 0.0

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// MAAT BUS
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
int dirCode = delta > 0 ? 1 : delta < 0 ? -1 : 0
float flowScore = f_clamp(buyPct + cvdZ * 3.0 - deceptionScore * 0.05, 0.0, 100.0)
int riskCode = anyClimax ? 5 : anyTrap ? 4 : anyAbsorb ? 3 : hiddenFlow ? 2 : relVol >= 1.5 ? 1 : 0
int eventCode = anyClimax ? 5 : anyAbsorb ? 6 : hiddenFlow ? 7 : relVol >= 1.5 ? 9 : dirCode != 0 ? 1 : 0
int eventDetailCode = 0
if buyClimaxTrap
    eventDetailCode := 33
else if sellClimaxTrap
    eventDetailCode := -33
else if buyClimaxExhaust
    eventDetailCode := 32
else if sellClimaxExhaust
    eventDetailCode := -32
else if buyClimaxStart
    eventDetailCode := 31
else if sellClimaxStart
    eventDetailCode := -31
else if buyAbsorb
    eventDetailCode := 21
else if sellAbsorb
    eventDetailCode := -21
else if deceptBuy
    eventDetailCode := 12
else if deceptSell
    eventDetailCode := -12
else if relVol >= 1.5
    eventDetailCode := 9
else
    eventDetailCode := dirCode

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// MAAT2 Compact BUS
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
bool tvValidData = not na(totalVol) and totalVol > 0.0 and not na(relVol)
float tvRvol = relVol
float tvDeltaRatio = tvValidData ? delta / totalVol : 0.0
bool tvDeltaBull = tvValidData and (tvDeltaRatio >= 0.15 or cvdZ >= 0.5)
bool tvDeltaBear = tvValidData and (tvDeltaRatio <= -0.15 or cvdZ <= -0.5)
float tvDryThreshold = 0.5
bool tvDry = tvValidData and relVol < tvDryThreshold
bool tvBuild = tvValidData and relVol >= 1.0 and relVol < 1.5
bool tvExpand = tvValidData and relVol >= 1.5
bool tvMixedConflict = tvValidData and (anyTrap or hiddenFlow or (isSmallBody and relVol >= absorbVolMult and math.abs(cvdZ) >= 0.8))

int tvDirPack = 1
if tvValidData
    if buyRatio >= 0.55 and delta > 0.0
        tvDirPack := 2
    else if sellRatio >= 0.55 and delta < 0.0
        tvDirPack := 0

int tvRiskCode = 0
if not tvValidData
    tvRiskCode := 5
else if anyClimax
    tvRiskCode := 4
else if tvMixedConflict
    tvRiskCode := 3
else if anyAbsorb or tvExpand
    tvRiskCode := 2
else if tvDry or tvBuild
    tvRiskCode := 1

int tvEventCode = 0
if not tvValidData
    tvEventCode := 9
else if anyClimax
    tvEventCode := 5
else if anyAbsorb
    tvEventCode := 4
else if tvMixedConflict
    tvEventCode := 8
else if tvDry
    tvEventCode := 1
else if tvExpand
    tvEventCode := 3
else if tvBuild
    tvEventCode := 2
else if tvDeltaBull
    tvEventCode := 6
else if tvDeltaBear
    tvEventCode := 7

int tvDetailCode = 0
if not tvValidData
    tvDetailCode := 90
else if buyClimax
    tvDetailCode := 50
else if sellClimax
    tvDetailCode := 51
else if sellAbsorb
    tvDetailCode := 40
else if buyAbsorb
    tvDetailCode := 41
else if tvMixedConflict
    tvDetailCode := 80
else if tvDry
    tvDetailCode := 10
else if tvExpand
    tvDetailCode := 30
else if tvBuild
    tvDetailCode := 20
else if tvDeltaBull
    tvDetailCode := 60
else if tvDeltaBear
    tvDetailCode := 61

float tvRvolScore = f_clamp(tvRvol / 3.0 * 35.0, 0.0, 35.0)
float tvEventBoost = anyClimax ? 25.0 : anyAbsorb ? 20.0 : tvMixedConflict ? 15.0 : tvExpand ? 14.0 : tvBuild ? 8.0 : tvDry ? 2.0 : 0.0
float tvDeltaScore = f_clamp(math.abs(tvDeltaRatio) * 100.0 * 0.15 + math.abs(cvdZ) * 4.0, 0.0, 20.0)
float tvStructureScore = f_clamp(bodyRatio * 12.0 + math.max(upperWickRatio, lowerWickRatio) * 8.0, 0.0, 20.0)
float tvNoisePenalty = tvDry and not anyAbsorb and not anyClimax ? 8.0 : 0.0
float tvScore = tvValidData ? f_clamp(tvRvolScore + tvEventBoost + tvDeltaScore + tvStructureScore + deceptionScore * 0.10 - tvNoisePenalty, 0.0, 100.0) : 0.0
int tvValid = tvValidData ? 1 : 0
int tvPackCode = tvDirPack * 10000 + tvRiskCode * 1000 + tvEventCode * 10 + tvValid

//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
// 07. View / Color
//━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
buyColor = input.color(color.new(#26A69A, 0), "매수 하단", group=grpView)
sellColor = input.color(color.new(#EF5350, 0), "매도 상단", group=grpView)
weakColor = input.color(color.new(color.gray, 60), "저거래량 회색", group=grpView)
cvdColor = input.color(color.new(color.yellow, 0), "CVD", group=grpView)
avgColor = input.color(color.new(color.white, 0), "평균선", group=grpView)
deceptSellCol = input.color(color.new(color.orange, 0), "숨은매도", group=grpView)
deceptBuyCol = input.color(color.new(color.aqua, 0), "숨은매수", group=grpView)
absorbCol = input.color(color.new(color.yellow, 85), "흡수 배경", group=grpView)
climaxCol = input.color(color.new(color.fuchsia, 82), "클라이맥스 배경", group=grpView)
trapCol = input.color(color.new(color.red, 82), "트랩 배경", group=grpView)
barWidth = input.int(3, "막대 두께", minval=1, maxval=5, group=grpView, display=display.none)
showAvg = input.bool(true, "평균선 표시", group=grpView, display=display.none)
dimWeak = input.bool(true, "저거래량 회색 처리", group=grpView, display=display.none)
weakRel = input.float(0.5, "저거래 기준", minval=0.1, step=0.1, group=grpView, display=display.none)
showPct = input.bool(true, "매수·매도 % 라벨 표시", group=grpView, display=display.none)
pctMinRel = input.float(1.0, "% 라벨 최소 상대거래량", minval=0.0, step=0.1, group=grpView, display=display.none)
showFlags = input.bool(true, "플래그 표시", group=grpView, display=display.none)
showBg = input.bool(true, "위험 배경 표시", group=grpView, display=display.none)
plotScaleOpt = input.string("M", "오른쪽 스케일 단위", options=["원본", "K", "M", "B"], group=grpView, display=display.none)

float sf = plotScaleOpt == "K" ? 1e3 : plotScaleOpt == "M" ? 1e6 : plotScaleOpt == "B" ? 1e9 : 1.0
bool weakBar = dimWeak and relVol < weakRel
color topCol = weakBar ? weakColor : sellColor
color botCol = weakBar ? weakColor : buyColor

plot(showVisuals ? totalVol / sf : na, title="매도(상단)", style=plot.style_columns, color=topCol, linewidth=barWidth, display=display.pane)
plot(showVisuals ? buyVol / sf : na, title="매수(하단)", style=plot.style_columns, color=botCol, linewidth=barWidth, display=display.pane)
plot(showVisuals and showAvg ? volAvg / sf : na, title="평균", color=avgColor, linewidth=1, display=display.pane)

float cvdBaseY = (volAvg * 0.15) / sf
float cvdDisp = (cvdZ * volAvg * cvdScale + volAvg * 0.15) / sf
plot(showVisuals and showCVD ? cvdDisp : na, title="CVD", color=cvdColor, linewidth=2, display=display.pane)
plot(showVisuals and showCVD ? cvdBaseY : na, title="CVD 0", color=color.new(color.yellow, 75), linewidth=1, display=display.pane)

plotchar(showVisuals and showFlags and deceptSell ? 1 : na, title="숨은매도", char="▼", location=location.top, color=deceptSellCol, size=size.tiny)
plotchar(showVisuals and showFlags and deceptBuy ? 1 : na, title="숨은매수", char="▲", location=location.bottom, color=deceptBuyCol, size=size.tiny)
plotchar(showVisuals and showFlags and buyAbsorb ? 1 : na, title="매수흡수", char="B", location=location.bottom, color=color.aqua, size=size.tiny)
plotchar(showVisuals and showFlags and sellAbsorb ? 1 : na, title="매도흡수", char="S", location=location.top, color=color.orange, size=size.tiny)

color bgCol = na
if showBg and anyTrap
    bgCol := trapCol
else if showBg and anyClimax
    bgCol := climaxCol
else if showBg and anyAbsorb
    bgCol := absorbCol
bgcolor(showVisuals ? bgCol : na)

if showVisuals and showPct and relVol >= pctMinRel and barstate.isconfirmed
